   853→   853→export function t(key: string, ...args: (string | number)[]): string {
   854→   854→  let text = dictionaries[currentLang][key] ?? dictionaries.fa[key] ?? key;
   855→   855→  args.forEach((arg, i) => {
   856→   856→    text = text.replace(`{${i}}`, String(arg));
   857→   857→  });
   858→   858→  return text;
   859→   859→}

   863→   863→/** تبدیل اعداد به فارسی */
   864→   864→export function faNum(n: number | string): string {
   865→   865→  const s = typeof n === 'number' ? String(Math.round(n)) : n;
   866→   866→  if (currentLang !== 'fa') return s;
   867→   867→  return s.replace(/[0-9]/g, (d) => FA_DIGITS[Number(d)]);
   868→   868→}
