   166→  start() {
   167→    this.lastFrame = performance.now();
   168→    const loop = (t: number) => {
   169→      if (this.disposed) return;
   170→      this.raf = requestAnimationFrame(loop);
   171→      const rawDt = Math.min(0.05, (t - this.lastFrame) / 1000);
   172→      this.lastFrame = t;
   173→      if (!this.paused && !this.gameOver) {
   174→        const dt = rawDt * this.speed;
   175→        this.update(dt);
   176→      }
   177→      this.cameraCtrl.update(rawDt);
   178→      this.scene3d.update(rawDt);
   179→      this.vfx.update(this.paused ? rawDt * 0.2 : rawDt * this.speed);
   180→      this.scene3d.render(this.cameraCtrl.camera);
   181→    };
   182→    this.raf = requestAnimationFrame(loop);
   183→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   184→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: 0 });
   185→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   186→    this.bus.emit('magic', this.magicState());
   187→    this.bus.emit('hero', this.heroState());
   188→  }

   190→  private update(dt: number) {
   191→    this.time += dt;
   192→
   193→    // موج — شمارش معکوس شروع خودکار
   194→    if (!this.waveActive && !this.gameOver) {
   195→      this.autoStartTimer -= dt;
   196→      if (this.autoStartTimer <= 0 && this.waveIndex < this.level.waves.length) {
   197→        if (this.waveIndex > 0 || !this.save.tutorialDone) {
   198→          // در آموزش شروع خودکار نداریم
   199→          if (this.save.tutorialDone) this.startWave();
   200→        } else {
   201→          this.startWave();
   202→        }
   203→      }
   204→      this.uiTimer -= dt;
   205→      if (this.uiTimer <= 0) {
   206→        this.uiTimer = 0.25;
   207→        this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: Math.max(0, this.autoStartTimer), earlyBonus: this.earlyBonus });
   208→      }
   209→    }
   210→
   211→    // اسپاون
   212→    if (this.waveActive) {
   213→      this.waveTime += dt;
   214→      while (this.spawnQueue.length && this.spawnQueue[0].time <= this.waveTime) {
   215→        const item = this.spawnQueue.shift()!;
   216→        this.spawnEnemy(item.enemyId, item.tier);
   217→      }
   218→    }
   219→
   220→    // دشمنان
   221→    const camPos = this.cameraCtrl.camera.position;
   222→    for (let i = this.enemies.length - 1; i >= 0; i--) {
   223→      const e = this.enemies[i];
   224→      e.update(dt, this.time);
   225→      e.updateLod(camPos);
   226→      if (!e.alive && e.getPos().y < -2.4) {
   227→        e.remove();
   228→        this.enemies.splice(i, 1);
   229→      }
   230→    }
   231→
   232→    // اتمام موج
   233→    if (this.waveActive && this.spawnQueue.length === 0 && this.enemies.length === 0 && !this.gameOver) {
   234→      this.completeWave();
   235→    }
   236→
   237→    // برج‌ها
   238→    for (const t of this.towers) {
   239→      t.update(dt, this.time, this.enemies);
   240→    }
   241→
   242→    // پرتابه‌ها
   243→    for (let i = this.projectiles.length - 1; i >= 0; i--) {
   244→      const p = this.projectiles[i];
   245→      p.update(dt, this.scene3d.scene);
   246→      if (!p.isAlive) this.projectiles.splice(i, 1);
   247→    }
   248→
   249→    // نبض پیش‌نمایش ساخت
   250→    if (this.ghostGroup && this.ghostGroup.visible) {
   251→      this.ghostPulse += dt * 4;
   252→      const pulse = 0.32 + Math.sin(this.ghostPulse) * 0.1;
   253→      const color = this.ghostValid ? 0x22c55e : 0xef4444;
   254→      (this.ghostDisc!.material as THREE.MeshBasicMaterial).opacity = pulse;
   255→      (this.ghostDisc!.material as THREE.MeshBasicMaterial).color.setHex(color);
   256→      (this.ghostRing!.material as THREE.MeshBasicMaterial).color.setHex(color);
   257→    }
   258→
   259→    // قهرمان
   260→    this.hero.update(dt, this.time);
   261→    this.hero.updateLod(this.cameraCtrl.camera.position);
   262→    // حلقه پالس حالت جابجایی — زیر قهرمان تا کاربر بداند کجا تپ کند
   263→    if (this.heroSelectMode && this.hero.alive) {
   264→      this.heroMovePulse -= dt;
   265→      if (this.heroMovePulse <= 0) {
   266→        this.heroMovePulse = 0.55;
   267→        this.vfx.ring(this.hero.getPos(), 0xffb020, 2.2, 0.5);
   268→      }
   269→    }
   270→    const commandBuff = this.hero.data.abilityKind === 'battleCommand' && this.hero.abilityActiveTime > 0
   271→      ? this.hero.data.abilityBuff / 100 : 0;
   272→    const blessingBuff = this.time < this.blessingUntil
   273→      ? (MAGIC.azarBlessing.rateBuffPct ?? 0) : 0;
   274→    this.ctx.towerRateBuff = commandBuff + blessingBuff;
   275→
   276→    // رگبار آتش/تیر
   277→    this.barrageTimer -= dt;
   278→    for (let i = this.fireBarrage.length - 1; i >= 0; i--) {
   279→      const item = this.fireBarrage[i];
   280→      item.time -= dt;
   281→      if (item.time <= 0) {
   282→        if (item.visual === 'arrow') {
   283→          const from = item.pos.clone().add(new THREE.Vector3((Math.random() - 0.5) * 8, 16, (Math.random() - 0.5) * 8));
   284→          this.projectiles.push(new Projectile('bolt', from, null, { damage: item.damage, color: 0xc084fc }, this.ctx, this.scene3d.scene));
   285→        } else {
   286→          const from = item.pos.clone().add(new THREE.Vector3(0, 22, 0));
   287→          this.projectiles.push(new Projectile('fireball', from, null, { damage: item.damage, splash: 2.6, burn: 4, color: 0xf97316 }, this.ctx, this.scene3d.scene));
   288→        }
   289→        this.fireBarrage.splice(i, 1);
   290→      }
   291→    }
   292→
   293→    // سردست‌های جادو
   294→    for (const id of Object.keys(this.magicCd)) {
   295→      this.magicCd[id] = Math.max(0, this.magicCd[id] - dt);
   296→    }
   297→
   298→    // سپر هسته
   299→    if (this.shieldUntil > 0 && this.time > this.shieldUntil) {
   300→      this.shieldUntil = 0;
   301→      this.bus.emit('shield', { active: false });
   302→    }
   303→
   304→    // بروزرسانی UI با نرخ محدود
   305→    this.uiTimer -= dt;
   306→    if (this.uiTimer <= 0) {
   307→      this.uiTimer = 0.2;
   308→      this.bus.emit('magic', this.magicState());
   309→      this.bus.emit('hero', this.heroState());
   310→      if (this.selectedTower) this.emitTowerInfo();
   311→    }
   312→  }
   313→
   314→  // ===================== موج‌ها =====================
   315→
   316→  startWave() {
   317→    if (this.waveActive || this.waveIndex >= this.level.waves.length || this.gameOver) return;
   318→    const wave = this.level.waves[this.waveIndex];
   319→    this.waveActive = true;
   320→    this.waveTime = 0;
   321→    this.spawnQueue = [];
   322→    for (const g of wave.groups) {
   323→      for (let i = 0; i < g.count; i++) {
   324→        this.spawnQueue.push({ enemyId: g.enemy, tier: g.tier, time: g.delay + i * g.interval });
   325→      }
   326→    }
   327→    this.spawnQueue.sort((a, b) => a.time - b.time);
   328→    this.earlyBonus = Math.round(15 + this.autoStartTimer * 4);
   329→    if (this.earlyBonus > 0 && this.autoStartTimer > 1) {
   330→      this.coins += this.earlyBonus;
   331→      this.coinsEarned += this.earlyBonus;
   332→      this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   333→    }
   334→    this.autoStartTimer = 0;
   335→
   336→    this.audio.play('waveStart');
   337→    this.bus.emit('banner', { wave: this.waveIndex + 1, boss: !!wave.boss });
   338→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: true, autoIn: 0 });
   339→
   340→    if (wave.boss) {
   341→      this.audio.setBossMode(true);
   342→      this.audio.play('bossRoar');
   343→      this.cameraCtrl.shake(0.6);
   344→      this.towersBuiltBeforeBoss = this.towersBuilt;
   345→    }
   346→
   347→    // پاداش کریستال موج
   348→    this.crystals += 5;
   349→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   350→
   351→    // آموزش
   352→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'upgrade');
   353→  }
   354→
   355→  private completeWave() {
   356→    const wave = this.level.waves[this.waveIndex];
   357→    this.waveActive = false;
   358→    this.waveIndex++;
   359→    this.save.stats.wavesSurvived = this.totalWavesSurvivedAtStart + this.waveIndex;
   360→    this.coins += wave.coinBonus;
   361→    this.coinsEarned += wave.coinBonus;
   362→    this.autoStartTimer = wave.boss ? 0 : 14;
   363→    this.earlyBonus = 0;
   364→    this.audio.play('reward');
   365→    this.bus.emit('waveCleared', { wave: this.waveIndex, bonus: wave.coinBonus });
   366→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   367→
   368→    if (this.waveIndex >= this.level.waves.length) {
   369→      this.doVictory();
   370→    } else {
   371→      this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: this.autoStartTimer, earlyBonus: 0 });
   372→      if (!this.save.tutorialDone) this.bus.emit('tutorial', 'magic');
   373→    }
   374→  }

   378→  private spawnEnemy(enemyId: string, tier: number) {
   379→    const data = ENEMIES[enemyId];
   380→    if (!data) return;
   381→    const enemy = new Enemy(data, tier, this.scene3d.scene, this.ctx);
   382→    this.enemies.push(enemy);
   383→    this.vfx.burst(enemy.getPos().clone().add(new THREE.Vector3(0, 1.5, 0)), data.accent, 10, { speed: 4, life: 0.5, size: 0.5, gravity: 3 });
   384→    if (data.boss) {
   385→      this.audio.play('bossRoar');
   386→      this.cameraCtrl.shake(0.7);
   387→    }
   388→  }
   389→
   390→  private summonShadow(pos: THREE.Vector3, enemyId = 'shadow') {
   391→    const data = ENEMIES[enemyId] ?? ENEMIES.shadow;
   392→    const e = new Enemy(data, 1, this.scene3d.scene, this.ctx);
   393→    // قرار دادن روی نزدیک‌ترین نقطه مسیر
   394→    let best = 0;
   395→    let bestD = Infinity;
   396→    this.scene3d.pathPoints.forEach((p, i) => {
   397→      const d = (p.x - pos.x) ** 2 + (p.z - pos.z) ** 2;
   398→      if (d < bestD) {
   399→        bestD = d;
   400→        best = i;
   401→      }
   402→    });
   403→    e.dist = (best / (this.scene3d.pathPoints.length - 1)) * this.scene3d.pathLength;
   404→    this.enemies.push(e);
   405→    this.vfx.burst(pos.clone().add(new THREE.Vector3(0, 1, 0)), data.accent, 14, { speed: 5, life: 0.6, size: 0.6, gravity: 3 });
   406→  }

   408→  /** سپر اثری آریا — مصونیت کامل هسته */
   409→  private requestCoreShield(duration: number) {
   410→    this.shieldUntil = Math.max(this.shieldUntil, this.time + duration);
   411→    this.bus.emit('shield', { active: true, duration });
   412→  }
   413→
   414→  private handleKill(e: Enemy) {
   415→    this.coins += e.reward;
   416→    this.coinsEarned += e.reward;
   417→    this.crystals += e.data.crystalReward ?? 0;
   418→    this.killsThisLevel++;
   419→    this.save.stats.kills++;
   420→    if (e.data.boss) {
   421→      this.save.stats.bossKills++;
   422→      this.audio.setBossMode(false);
   423→    }
   424→    this.hero.gainXp(e.data.boss ? 200 : 12);
   425→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   426→  }
   427→
   428→  private handleLeak(e: Enemy) {
   429→    // حذف از صحنه
   430→    e.alive = false;
   431→    e.remove();
   432→    const i = this.enemies.indexOf(e);
   433→    if (i >= 0) this.enemies.splice(i, 1);
   434→
   435→    if (this.time < this.shieldUntil) {
   436→      // سپر آریانا — جلوگیری کامل
   437→      this.vfx.ring(this.scene3d.corePos, 0xfbbf24, 6, 0.5);
   438→      return;
   439→    }
   440→    this.leaks++;
   441→    this.coreHp = Math.max(0, this.coreHp - e.data.coreDamage);
   442→    this.cameraCtrl.shake(e.data.boss ? 1 : 0.45);
   443→    this.vfx.explosion(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 2, 0)), 0xef4444, e.data.boss);
   444→    this.audio.play('explosion');
   445→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   446→    if (this.coreHp <= 0 && !this.gameOver) {
   447→      this.doDefeat();
   448→    }
   449→  }

   453→  /** از کار انداختن موقت برج توسط توانایی فاز ۲ Boss */
   454→  disableTower(towerId: number, duration: number, color = 0xf97316) {
   455→    const t = this.towers.find((x) => x.id === towerId);
   456→    if (!t || !t.alive) return;
   457→    t.disable(duration, color);
   458→  }

   460→  setBuildMode(towerId: string | null) {
   461→    // قفل برج — باز شدن تدریجی با ستاره
   462→    if (towerId && !isTowerUnlocked(this.save, towerId)) {
   463→      this.audio.play('error');
   464→      this.bus.emit('toast', { key: 'towerLocked', type: 'error' });
   465→      return;
   466→    }
   467→    this.buildMode = towerId;
   468→    this.magicPending = null;
   469→    this.heroSelectMode = false;

   576→  tryBuild(towerId: string, point: THREE.Vector3): boolean {
   577→    const data = TOWERS[towerId];
   578→    if (!data) return false;
   579→    // قفل برج — حتی با تپ مستقیم روی زمین
   580→    if (!isTowerUnlocked(this.save, towerId)) {
   581→      this.audio.play('error');
   582→      this.bus.emit('toast', { key: 'towerLocked', type: 'error' });
   583→      return false;
   584→    }
   585→    const x = Math.round(point.x / 2) * 2;
   586→    const z = Math.round(point.z / 2) * 2;
   587→    if (!this.scene3d.canBuild(x, z, this.towers.map((t) => t.position))) {
   588→      this.audio.play('error');
   589→      this.bus.emit('toast', { key: 'invalidSpot', type: 'error' });
   590→      return false;
   591→    }
   592→    if (this.coins < data.cost) {
   593→      this.audio.play('error');
   594→      this.bus.emit('toast', { key: 'notEnoughCoins', type: 'error' });
   595→      return false;
   596→    }
   597→    this.coins -= data.cost;
   598→    const tower = new Tower(this.towerNextId++, towerId, new THREE.Vector3(x, 0, z), this.scene3d.scene, this.ctx);
   599→    this.towers.push(tower);
   600→    this.towersBuilt++;
   601→    this.save.stats.towersBuilt++;
   602→    this.audio.play('place');
   603→    this.vfx.ring(tower.position, data.color, 4, 0.5);
   604→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   605→    // خروج از حالت ساخت — هماهنگ با رویداد buildMode
   606→    this.buildMode = null;
   607→    this.clearGhost();
   608→    this.setBuildGridVisible(false);
   609→    this.bus.emit('buildMode', { towerId: null });
   610→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'wave');
   611→    return true;
   612→  }

   828→  private chainLightning(from: THREE.Vector3, first: DamageableEnemy, chains: number, damage: number, color: number) {
   829→    const hitList: DamageableEnemy[] = [];
   830→    let current = first;
   831→    let currentDamage = damage;
   832→    let currentFrom = from;
   833→    for (let i = 0; i <= chains; i++) {
   834→      if (!current || !current.alive) break;
   835→      this.vfx.bolt(currentFrom, current.getPos().clone().add(new THREE.Vector3(0, 1.2, 0)), color);
   836→      hitList.push(current);
   837→      current.takeDamage(currentDamage, 'energy');
   838→      currentDamage *= 0.72;
   839→      currentFrom = current.getPos().clone().add(new THREE.Vector3(0, 1, 0));
   840→      // هدف بعدی
   841→      let next: DamageableEnemy | null = null;
   842→      let bestD = 8;
   843→      for (const e of this.enemies) {
   844→        if (!e.alive || hitList.includes(e)) continue;
   845→        const d = e.getPos().distanceTo(currentFrom);
   846→        if (d < bestD) {
   847→          bestD = d;
   848→          next = e;
   849→        }
   850→      }
   851→      current = next as DamageableEnemy;
   852→      if (!next) break;
   853→    }
   854→  }

   871→  private handleTap(hit: TapHit) {
   872→    if (this.gameOver) return;
   873→
   874→    // انتخاب شیء؟
   875→    if (hit.objects.length > 0) {
   876→      const root = hit.objects[0].object;
   877→      // برج؟
   878→      const tower = this.towers.find((t) => {
   879→        let o: THREE.Object3D | null = root;
   880→        while (o) {
   881→          if (o === t.mesh) return true;
   882→          o = o.parent;
   883→        }
   884→        return false;
   885→      });
   886→      if (tower) {
   887→        this.selectedTower = tower;
   888→        this.emitTowerInfo();
   889→        this.buildMode = null;
   890→        this.heroSelectMode = false;
   891→        this.clearGhost();
   892→        this.setBuildGridVisible(false);
   893→        this.audio.play('tap');
   894→        return;
   895→      }
   896→      // قهرمان؟ — تپ روی هیرو = ورود به حالت جابجایی
   897→      if (root === this.hero.mesh || (root.parent && root.parent === this.hero.mesh)) {
   898→        if (this.hero.alive) {
   899→          this.buildMode = null;
   900→          this.clearGhost();
   901→          this.setBuildGridVisible(false);
   902→          this.selectHeroMode(true);
   903→          this.bus.emit('heroSelected', {});
   904→          this.bus.emit('toast', { key: 'heroMoveHint', type: 'info' });
   905→        } else {
   906→          this.bus.emit('heroSelected', {});
   907→        }
   908→        this.audio.play('tap');
   909→        return;
   910→      }
   911→    }
   912→
   913→    // جادو در انتظار هدف
   914→    if (this.magicPending) {
   915→      const id = this.magicPending;
   916→      this.magicPending = null;
   917→      this.heroSelectMode = false;
   918→      this.bus.emit('magicPending', { id: null });
   919→      this.castMagic(id, hit.ground);
   920→      return;
   921→    }
   922→
   923→    // حالت ساخت — اول پیش‌نمایش در نقطه تپ، بعد ساخت
   924→    if (this.buildMode) {
   925→      this.updateGhost(hit.ground);
   926→      this.tryBuild(this.buildMode, hit.ground);
   927→      return;
   928→    }
   929→
   930→    // حرکت/استقرار قهرمان
   931→    if (this.heroSelectMode) {
   932→      if (!this.hero.deployed) {
   933→        this.deployHero(hit.ground);
   934→      } else {
   935→        this.hero.moveTo(hit.ground);
   936→        this.vfx.ring(hit.ground, 0xffb020, 1.5, 0.4);
   937→      }
   938→      this.selectHeroMode(false);
   939→      return;
   940→    }
   941→
   942→    this.selectedTower = null;
   943→    this.bus.emit('selectTower', null);
   944→  }

   947→  deployHero(point: THREE.Vector3) {

   962→  private doVictory() {
   963→    this.gameOver = true;
   964→    const healthPct = (this.coreHp / this.coreMaxHp) * 100;
   965→    const star1 = true;
   966→    const star2 = healthPct >= this.level.star2HealthPct;
   967→    const star3 = this.leaks === 0;
   968→    const stars = (star1 ? 1 : 0) + (star2 ? 1 : 0) + (star3 ? 1 : 0);
   969→
   970→    // پاداش‌ها
   971→    const rewardCrystals = 10 + stars * 5;
   972→    const rewardKnowledge = 2 + stars;
   973→    this.crystals += rewardCrystals;
   974→    this.save.crystals = this.crystals;
   975→    this.save.knowledge += rewardKnowledge;
   976→    this.save.stats.victories++;
   977→    if (this.leaks === 0 && !this.save.achievements.includes('untouchable')) this.save.achievements.push('untouchable');
   978→    if (this.coreHp === this.coreMaxHp && !this.save.achievements.includes('perfect_defense')) this.save.achievements.push('perfect_defense');
   979→    if (healthPct < 10 && !this.save.achievements.includes('last_stand')) this.save.achievements.push('last_stand');
   980→
   981→    const prev = this.save.levels[this.level.id];
   982→    const bestStars = Math.max(prev?.stars ?? 0, stars);
   983→    this.save.levels[this.level.id] = {
   984→      stars: bestStars,
   985→      bestHealthPct: Math.max(prev?.bestHealthPct ?? 0, Math.round(healthPct)),
   986→      completed: true,
   987→    };
   988→    // بازمحاسبه مجموع ستاره‌ها
   989→    this.save.stats.totalStars = Object.values(this.save.levels).reduce((sum, l) => sum + l.stars, 0);
   990→    this.save.tutorialDone = true;
   991→
   992→    this.audio.play('victory');
   993→    this.vfx.burst(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 3, 0)), 0xfbbf24, 80, { speed: 10, life: 1.4, size: 0.8, gravity: 4, up: 1 });
   994→    this.bus.emit('victory', {
   995→      stars,
   996→      healthPct: Math.round(healthPct),
   997→      rewards: { crystals: rewardCrystals, knowledge: rewardKnowledge, coins: this.coinsEarned },
   998→    });
   999→  }
