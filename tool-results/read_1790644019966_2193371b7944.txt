   497→function EndOverlay({ title, titleClass, subtitle, stars, rewards, actions }: {
   498→  title: string;
   499→  titleClass: string;
   500→  subtitle: string;
   501→  stars: number;
   502→  rewards?: { icon: React.ComponentType<{ className?: string; style?: React.CSSProperties }>; value: string; label: string; color: string }[];
   503→  actions: BattleActions;
   504→}) {
   505→  const backToMap = () => {
   506→    actions.endBattle();
   507→    useGameStore.getState().setScreen('map');
   508→  };
   509→  const restart = () => {
   510→    actions.endBattle();
   511→    setTimeout(() => actions.startBattle(), 60);
   512→  };
   513→
   514→  return (
   515→    <div className="absolute inset-0 z-40 flex items-center justify-center bg-black/85 backdrop-blur-sm pointer-events-auto">
   516→      <div className="w-full max-w-sm rounded-3xl border border-amber-400/30 bg-[#0d1420] p-6 text-center end-anim">
   517→        <h2 className={`bg-gradient-to-b bg-clip-text text-4xl font-black text-transparent ${titleClass}`}>
   518→          {title}
   519→        </h2>
   520→        <p className="mt-2 text-sm font-bold text-slate-400">{subtitle}</p>
   521→
   522→        {stars > 0 && (
   523→          <div className="mt-4 flex justify-center gap-2">
   524→            {[0, 1, 2].map((i) => (
   525→              <Star key={i}
   526→                className={`h-12 w-12 star-pop ${i < stars ? 'fill-amber-400 text-amber-400 drop-shadow-[0_0_12px_rgba(245,158,11,0.7)]' : 'text-slate-700'}`}
   527→                style={{ animationDelay: `${i * 0.35}s` }} />
   528→            ))}
   529→          </div>
   530→        )}
   531→
   532→        {rewards && (
   533→          <div className="mt-5">
   534→            <div className="text-xs font-black text-slate-500">{t('rewards')}</div>
   535→            <div className="mt-2 flex justify-center gap-4">

   538→                  <r.icon className={`h-5 w-5 ${r.color}`} />

   556→            {t('quitToMap')}
   557→          </button>
