   857→let currentLang: Language = 'fa';
   858→
   859→export function setLanguage(lang: Language) {
   860→  currentLang = lang;
   861→}
   862→
   863→export function getLanguage(): Language {
   864→  return currentLang;
   865→}

   867→export function t(key: string, ...args: (string | number)[]): string {
   868→  let text = dictionaries[currentLang][key] ?? dictionaries.fa[key] ?? key;
   869→  args.forEach((arg, i) => {
   870→    text = text.replace(`{${i}}`, String(arg));
   871→  });
   872→  return text;
   873→}

   875→const FA_DIGITS = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
   876→
   877→/** تبدیل اعداد به فارسی */
   878→export function faNum(n: number | string): string {
   879→  const s = typeof n === 'number' ? String(Math.round(n)) : n;
   880→  if (currentLang !== 'fa') return s;
   881→  return s.replace(/[0-9]/g, (d) => FA_DIGITS[Number(d)]);
   882→}
