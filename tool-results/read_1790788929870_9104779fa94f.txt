   137→   137→/** PRNG قطعی — هر مرحله ترکیب ثابت و تکرارپذیر خودش را دارد */
   138→   138→function mulberry32(seed: number): () => number {
   139→   139→  let s = seed >>> 0;
   140→   140→  return () => {
   141→   141→    s = (s + 0x6d2b79f5) >>> 0;
   142→   142→    let t = s;
   143→   143→    t = Math.imul(t ^ (t >>> 15), t | 1);
   144→   144→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   145→   145→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   146→   146→  };
   147→   147→}

   204→   204→/** بخش ۲۱ — HP مؤثر با زره flat (ضربه نماینده ۴۵) — برای نرمال‌سازی موج */
   205→   205→function unitEff(enemyId: string, tier: number): number {
   206→   206→  const d = ENEMIES[enemyId];
   207→   207→  if (!d) return 0;
   208→   208→  const hp = tiered(d, tier).hp;
   209→   209→  return hp * (45 / Math.max(8, 45 - d.armor * 0.7));
   210→   210→}

   212→   212→function waveEffHp(groups: SpawnGroup[]): number {
   213→   213→  let sum = 0;
   214→   214→  for (const g of groups) sum += g.count * unitEff(g.enemy, g.tier);
   215→   215→  return sum;
   216→   216→}

   218→   218→/** بخش ۲۱ — نرمال‌سازی صحیح موج به منحنی هدف
   219→   219→ * هر گام: تک‌حرکتی که eff را بیشترین نزدیکی به هدف می‌برد (همگرایی یکنوا، بدون نوسان)
   220→   220→ * escortMin: کف تعداد محافظ‌ها در موج Boss (بخش ۲۱ — Boss Combination) */
   221→   221→function normalizeWave(groups: SpawnGroup[], target: number, skipBoss = false, escortMin = 0) {
   222→   222→  const LO = target * 0.9, HI = target * 1.1;
   223→   223→  for (let guard = 0; guard < 240; guard++) {
   224→   224→    const eff = waveEffHp(groups);
   225→   225→    if (eff > HI) {
   226→   226→      let best = -1, bestErr = Math.abs(eff - target);
   227→   227→      groups.forEach((g, i) => {
   228→   228→        if (g.count <= 1) return;
   229→   229→        if (skipBoss && ENEMIES[g.enemy]?.boss) return;
   230→   230→        if (escortMin > 0 && g.count <= escortMin) return;
   231→   231→        const err = Math.abs(eff - unitEff(g.enemy, g.tier) - target);
   232→   232→        if (err < bestErr) { bestErr = err; best = i; }
   233→   233→      });
   234→   234→      if (best < 0) break;
   235→   235→      groups[best].count--;
   236→   236→    } else if (eff < LO) {
   237→   237→      let best = -1, bestErr = Math.abs(eff - target);
   238→   238→      groups.forEach((g, i) => {
   239→   239→        if (skipBoss && ENEMIES[g.enemy]?.boss) return;
   240→   240→        if (g.count >= 40) return; // بخش ۲۱ — ضد یکنواختی، گروه‌های بزرگ‌تر پر نمی‌شوند
   241→   241→        const err = Math.abs(eff + unitEff(g.enemy, g.tier) - target);
   242→   242→        if (err < bestErr) { bestErr = err; best = i; }
   243→   243→      });
   244→   244→      if (best < 0) break;
   245→   245→      groups[best].count++;
   246→   246→    } else break;
   247→   247→  }
   248→   248→}

   250→   250→function buildWaves(chapter: ChapterDef, stage: number, pathCount: number): WaveData[] {
   251→   251→  const D = (chapter.id - 1) * 10 + stage; // شاخص سختی ۱..۱۰۰
   252→   252→  const rng = mulberry32(chapter.id * 7919 + stage * 104729);
   253→   253→  const pool = CHAPTER_POOLS[chapter.id];
   254→   254→  const comp = POOL_COMP[chapter.id];
   255→   255→  // بخش ۲۱ — tier پیوسته: HP نرم رشد می‌کند (بدون پرتگاه ×۱٫۶۵ در مرز فصل)
   256→   256→  const tFrac = Math.round((1 + D * 0.0265) * 100) / 100;
   257→   257→  const tierMult = 1 + 0.65 * (tFrac - 1);
   258→   258→  const intensity = 0.8 + D * 0.006; // ۰٫۸ تا ۱٫۴
   259→   259→  const waves: WaveData[] = [];
   260→   260→
   261→   261→  for (let w = 0; w < 10; w++) {
   262→   262→    // ----- موج Boss -----
   263→   263→    if (w === 9) {
   264→   264→      const bossTier = Math.round((1 + (D - 1) * 0.024) * 100) / 100;
   265→   265→      const groups: SpawnGroup[] = [
   266→   266→        { enemy: chapter.bossId, count: 1, interval: 1, delay: 2, tier: bossTier, path: 0 },
   267→   267→      ];
   268→   268→      // محافظ‌های Boss — پخش روی همه مسیرها در مراحل چندمسیره
   269→   269→      const escorts = pool.filter((e) => roleOf(e) !== 'utility');
   270→   270→      const e1 = escorts[Math.floor(rng() * escorts.length)];
   271→   271→      const e2 = escorts[Math.floor(rng() * escorts.length)];
   272→   272→      const eCount = Math.round(6 + D * 0.06);
   273→   273→      groups.push({ enemy: e1, count: eCount, interval: Math.max(1.4, 3.2 - D * 0.012), delay: 12, tier: tFrac });
   274→   274→      if (D > 35) {
   275→   275→        groups.push({ enemy: e2, count: Math.max(3, Math.round(eCount * 0.6)), interval: Math.max(1.8, 3.6 - D * 0.012), delay: 20, tier: tFrac });
   276→   276→      }
   277→   277→      // بخش ۲۱ — نرمال‌سازی موج Boss (Boss ثابت؛ محافظ‌ها با کف ۴۰٪ حفظ می‌شوند)
   278→   278→      const bossTarget = WAVE_BASE * growAt(D) * intensity * 1.23 * tierMult * comp * 1.55;
   279→   279→      normalizeWave(groups, bossTarget, true, Math.max(2, Math.floor(eCount * 0.4)));
   280→   280→      waves.push({
   281→   281→        coinBonus: Math.round(230 + D * 2.3),
   282→   282→        crystalBonus: 12 + chapter.id,
   283→   283→        groups: distributeGroups(groups, pathCount, rng),
   284→   284→        boss: true,
   285→   285→      });
   286→   286→      continue;
   287→   287→    }
   288→   288→
   289→   289→    // ----- موج‌های عادی -----
   290→   290→    const wavePower = intensity * (0.75 + w * 0.06);
   291→   291→    const nGroups = w < 2 ? 1 : w < 5 ? 2 : D > 40 ? 3 : 2;
   292→   292→    const picked: string[] = [];
   293→   293→    const poolShuffled = [...pool].sort(() => rng() - 0.5);
   294→   294→    // تنوع: هوایی از موج ۴ به بعد، شفادهنده/احضارکننده از موج ۵ به بعد، جادوگر از موج ۲
   295→   295→    for (const e of poolShuffled) {
   296→   296→      if (picked.length >= nGroups) break;
   297→   297→      const role = roleOf(e);
   298→   298→      if (role === 'air' && w < 3) continue;
   299→   299→      if (role === 'utility' && w < 4) continue;
   300→   300→      if (role === 'caster' && w < 1) continue; // موج اول بدون سپر جادوگر — شروع نرم
   301→   301→      if (picked.includes(e)) continue;
   302→   302→      picked.push(e);
   303→   303→    }
   304→   304→    // اگر نقش هوایی موج اول بود جایگزین زمینی
   305→   305→    while (picked.length < nGroups) {
   306→   306→      const fallback = poolShuffled.find((e) => !picked.includes(e) && roleOf(e) !== 'air');
   307→   307→      if (!fallback) break;
   308→   308→      picked.push(fallback);
   309→   309→    }
   310→   310→
   311→   311→    const groups: SpawnGroup[] = [];
   312→   312→    picked.forEach((enemyId, gi) => {
   313→   313→      const role = roleOf(enemyId);
   314→   314→      const base = ROLE_BASE[role];
   315→   315→      // بخش ۲۱ — jitter مهارشده (قبلاً ۰٫۸۵–۱٫۱۵ نوسان ±۵۰٪ بین مراحل مجاور می‌ساخت)
   316→   316→      const count = Math.max(1, Math.round(base.count * wavePower * (0.9 + rng() * 0.2)));
   317→   317→      const interval = Math.max(0.5, base.interval - D * 0.004 + (rng() - 0.5) * 0.3);
   318→   318→      const gTier = Math.round((tFrac + (w >= 6 && rng() < 0.3 ? 0.3 : 0)) * 100) / 100;
   319→   319→      groups.push({
   320→   320→        enemy: enemyId,
   321→   321→        count,
   322→   322→        interval: Math.round(interval * 10) / 10,
   323→   323→        delay: gi * (3 + rng() * 3),
   324→   324→        tier: gTier,
   325→   325→      });
   326→   326→    });
   327→   327→
   328→   328→    // بخش ۲۱ — نرمال‌سازی HP مؤثر موج به منحنی هدف
   329→   329→    // (ترکیب نقش‌ها و هویت دشمنان حفظ می‌شود؛ فقط فشار کل یکنواخت می‌گردد)
   330→   330→    const target = WAVE_BASE * growAt(D) * wavePower * tierMult * comp;
   331→   331→    normalizeWave(groups, target);
   332→   332→
   333→   333→    waves.push({
   334→   334→      coinBonus: Math.round(52 + D * 0.9 + w * 6),
   335→   335→      groups: distributeGroups(groups, pathCount, rng),
   336→   336→    });
   337→   337→  }
   338→   338→  return waves;
   339→   339→}
   340→   340→
   341→   341→/** پخش گروه‌ها روی مسیرها — در مراحل چندمسیره فشار متوازن می‌آید */
   342→   342→function distributeGroups(groups: SpawnGroup[], pathCount: number, rng: () => number): SpawnGroup[] {
   343→   343→  if (pathCount <= 1) return groups;
   344→   344→  const out: SpawnGroup[] = [];
   345→   345→  for (const g of groups) {
   346→   346→    // هر گروه به‌جز شفادهنده/احضارکننده — بین همه مسیرها تقسیم می‌شود تا هر مسیر فعال بماند
   347→   347→    if (g.count >= pathCount * 2 && roleOf(g.enemy) !== 'utility') {
   348→   348→      const per = Math.floor(g.count / pathCount);
   349→   349→      let rem = g.count % pathCount;
   350→   350→      for (let p = 0; p < pathCount; p++) {
   351→   351→        const c = per + (rem-- > 0 ? 1 : 0);
   352→   352→        if (c <= 0) continue;
   353→   353→        out.push({ ...g, count: c, path: p, delay: g.delay + p * (0.8 + rng() * 0.6) });
   354→   354→      }
   355→   355→    } else {
   356→   356→      // گروه خیلی کوچک — یک مسیر چرخشی
   357→   357→      out.push({ ...g, path: Math.floor(rng() * pathCount) });
   358→   358→    }
   359→   359→  }
   360→   360→  return out;
   361→   361→}

   365→   365→export function stageId(chapter: number, stage: number): string {
   366→   366→  return `c${chapter}s${stage}`;
   367→   367→}

   369→   369→/** درآمد کل مرحله بدون جبران (سکه شروع + کشتار + پاداش پایه) — پایه جبران اقتصاد (بخش ۲۰/۲۱) */
   370→   370→function stageTotalIncome(chapter: ChapterDef, stage: number): number {
   371→   371→  const lv = buildStage(chapter, stage, 1);
   372→   372→  let kill = 0, bonus = 0;
   373→   373→  for (const w of lv.waves) {
   374→   374→    bonus += w.coinBonus ?? 0;
   375→   375→    for (const g of w.groups) kill += g.count * (tiered(ENEMIES[g.enemy], g.tier)?.reward ?? 0);
   376→   376→  }
   377→   377→  return lv.startCoins + kill + bonus;
   378→   378→}
   379→   379→
   380→   380→export function buildStage(chapter: ChapterDef, stage: number, incomeComp = 1): LevelData {
   381→   381→  const D = (chapter.id - 1) * 10 + stage;
   382→   382→  const pathCount = pathCountFor(chapter.id, stage);
   383→   383→  const waves = buildWaves(chapter, stage, pathCount);
   384→   384→  if (incomeComp !== 1) {
   385→   385→    // بخش ۲۰ — موج‌های کم‌درآمد (تانک‌سنگین) پاداش بیشتر می‌گیرند تا اقتصاد صاف بماند
   386→   386→    for (const w of waves) w.coinBonus = Math.round((w.coinBonus ?? 0) * incomeComp);
   387→   387→  }
   388→   388→  return {
   389→   389→    id: stageId(chapter.id, stage),
   390→   390→    chapter: chapter.id,
   391→   391→    stage,
   392→   392→    nameKey: chapter.nameKey,
   393→   393→    descKey: chapter.descKey,
   394→   394→    coreHp: Math.round(100 + (chapter.id - 1) * 6 + (stage - 1) * 2),
   395→   395→    startCoins: Math.round(220 + chapter.id * 14 + stage * 6),
   396→   396→    startCrystals: Math.round(28 + chapter.id * 2 + (stage - 1) * 0.5),
   397→   397→    timeOfDay: chapter.timeOfDay,
   398→   398→    env: chapter.env,
   399→   399→    star2HealthPct: 70,
   400→   400→    star3ChallengeKey: 'starChallenge',
   401→   401→    paths: pickPaths(chapter.id, stage, pathCount),
   402→   402→    waves,
   403→   403→  };
   404→   404→}
   405→   405→
   406→   406→/** هر ۱۰۰ مرحله — یک بار در load ساخته می‌شود (داده خالص، هزینه نزدیک صفر)
   407→   407→ * دومرحله‌ای: اول درآمد کل هر مرحله، بعد هموارسازی به منحنی درآمد کمپین
   408→   408→ * (لنگر = میانگین فصل ۱؛ موج‌های کم‌درآمدِ تانک‌سنگین پاداش بیشتر می‌گیرند) */
   409→   409→function buildAllStages(): Record<string, LevelData> {
   410→   410→  const incomes: number[] = [];
   411→   411→  for (const ch of CHAPTERS) {
   412→   412→    for (let s = 1; s <= 10; s++) incomes.push(stageTotalIncome(ch, s));
   413→   413→  }
   414→   414→  const ref = incomes.slice(0, 10).reduce((a, r) => a + r, 0) / 10; // لنگر فصل ۱
   415→   415→  const levels: Record<string, LevelData> = {};
   416→   416→  let i = 0;
   417→   417→  for (const ch of CHAPTERS) {
   418→   418→    for (let s = 1; s <= 10; s++) {
   419→   419→      const D = (ch.id - 1) * 10 + s;
   420→   420→      const target = ref * Math.pow(1.0117, D - 1); // منحنی درآمد کمپین
   421→   421→      const incomeComp = Math.min(2.0, Math.max(0.6, target / incomes[i++]));
   422→   422→      levels[stageId(ch.id, s)] = buildStage(ch, s, incomeComp);
   423→   423→    }
   424→   424→  }
   425→   425→  return levels;
   426→   426→}

   432→   432→/** مرحله آزاد است؟ — زنجیره‌ای: مرحله قبلی یا Boss فصل قبل */
   433→   433→export function isStageUnlocked(chapter: number, stage: number, completed: Set<string>): boolean {
   434→   434→  if (chapter === 1 && stage === 1) return true;
   435→   435→  if (stage > 1) return completed.has(stageId(chapter, stage - 1));
   436→   436→  return completed.has(stageId(chapter - 1, 10));
   437→   437→}

   448→   448→export function firstPlayableStageId(save: SaveData): string {
   449→   449→  const completed = completedStageIds(save);
   450→   450→  for (const ch of CHAPTERS) {
   451→   451→    for (let s = 1; s <= 10; s++) {
   452→   452→      if (!completed.has(stageId(ch.id, s)) && isStageUnlocked(ch.id, s, completed)) {
   453→   453→        return stageId(ch.id, s);
   454→   454→      }
   455→   455→    }
   456→   456→  }
   457→   457→  return stageId(10, 10);
   458→   458→}
