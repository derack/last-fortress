   351→function TowerPanel({ actions }: { actions: BattleActions }) {
   352→  const st = useGameStore();
   353→  const tw = TOWERS[st.selectedTower!.towerId];
   354→  const game = actions.get();
   355→  const info = st.selectedTower!;
   356→  const Icon = TOWER_ICONS[info.towerId] ?? Crosshair;
   357→  const maxed = info.upgradeCost < 0;
   358→
   359→  return (
   360→    <div className="absolute inset-x-3 bottom-20 z-20 rounded-2xl border border-slate-600/50 bg-black/85 p-3 backdrop-blur-md pointer-events-auto panel-anim">
   361→      <div className="flex items-center gap-2">
   362→        <div className="flex h-11 w-11 items-center justify-center rounded-xl border"
   363→          style={{ borderColor: '#' + tw.color.toString(16).padStart(6, '0'), background: 'rgba(0,0,0,0.5)' }}>
   364→          <Icon className="h-6 w-6" style={{ color: '#' + tw.color.toString(16).padStart(6, '0') }} />
   365→        </div>
   366→        <div className="flex-1">
   367→          <div className="flex items-center gap-2">
   368→            <span className="text-base font-black text-slate-100">{t(tw.nameKey)}</span>
   369→            <span className="rounded-full bg-amber-400/20 px-2 py-0.5 text-[10px] font-black text-amber-300">
   370→              {t('level')} {faNum(info.level)}
   371→            </span>
   372→          </div>
   373→          <div className="mt-0.5 flex gap-3 text-[11px] font-bold text-slate-400">
   374→            <span>{t('damage')}: <b className="text-slate-200">{faNum(info.damage)}</b></span>
   375→            <span>{t('range')}: <b className="text-slate-200">{faNum(info.range)}</b></span>
   376→            <span>{t('fireRate')}: <b className="text-slate-200">{faNum(info.fireRate)}</b></span>
   377→          </div>
   378→        </div>
   379→        <button onClick={() => useGameStore.getState().handleEvent('selectTower', null)}
   380→          className="flex h-8 w-8 items-center justify-center rounded-lg bg-black/50 text-slate-400 active:scale-95">
   381→          <X className="h-4 w-4" />
   382→        </button>
   383→      </div>
   384→
   385→      <div className="mt-2.5 flex gap-2">
   386→        <button onClick={() => game?.upgradeTower(info.id)} disabled={maxed || st.coins < info.upgradeCost}
   387→          className={`flex flex-1 items-center justify-center gap-1.5 rounded-xl px-3 py-2.5 text-sm font-black active:scale-95
   388→            ${maxed ? 'border border-slate-700 bg-black/40 text-slate-500' :
   389→            st.coins >= info.upgradeCost ? 'bg-gradient-to-b from-emerald-500 to-emerald-700 text-white' : 'bg-slate-800 text-slate-500'}`}>
   390→          <ArrowUpCircle className="h-4 w-4" />
   391→          {maxed ? t('maxLevel') : `${t('upgrade')} — ${faNum(info.upgradeCost)}`}
   392→        </button>
   393→        <button onClick={() => game?.useTowerAbility(info.id)} disabled={info.abilityCd > 0}
   394→          className={`relative flex w-24 items-center justify-center gap-1 rounded-xl px-2 py-2.5 text-xs font-black active:scale-95

   429→function EndOverlay({ title, titleClass, subtitle, stars, rewards, actions }: {
   430→  title: string;
   431→  titleClass: string;
   432→  subtitle: string;
   433→  stars: number;
   434→  rewards?: { icon: React.ComponentType<{ className?: string; style?: React.CSSProperties }>; value: string; label: string; color: string }[];
   435→  actions: BattleActions;
   436→}) {
   437→  const backToMap = () => {
   438→    actions.endBattle();
   439→    useGameStore.getState().setScreen('map');
   440→  };
   441→  const restart = () => {
   442→    actions.endBattle();
   443→    setTimeout(() => actions.startBattle(), 60);

   445→
   446→  return (
   447→    <div className="absolute inset-0 z-40 flex items-center justify-center bg-black/85 backdrop-blur-sm pointer-events-auto">
   448→      <div className="w-full max-w-sm rounded-3xl border border-amber-400/30 bg-[#0d1420] p-6 text-center end-anim">

   450→          {title}
   451→        </h2>
   452→        <p className="mt-2 text-sm font-bold text-slate-400">{subtitle}</p>
   453→
   454→        {stars > 0 && (
   455→          <div className="mt-4 flex justify-center gap-2">
   456→            {[0, 1, 2].map((i) => (
   457→              <Star key={i}
   458→                className={`h-12 w-12 star-pop ${i < stars ? 'fill-amber-400 text-amber-400 drop-shadow-[0_0_12px_rgba(245,158,11,0.7)]' : 'text-slate-700'}`}
   459→                style={{ animationDelay: `${i * 0.35}s` }} />

   463→
   464→        {rewards && (
   465→          <div className="mt-5">
   466→            <div className="text-xs font-black text-slate-500">{t('rewards')}</div>

   487→            <Home className="h-5 w-5" />
   488→            {t('quitToMap')}
   489→          </button>
   490→        </div>
   491→      </div>
   492→    </div>
   493→  );
   494→}
