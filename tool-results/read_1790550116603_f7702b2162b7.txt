   149→  start() {
   150→    this.lastFrame = performance.now();
   151→    const loop = (t: number) => {
   152→      if (this.disposed) return;
   153→      this.raf = requestAnimationFrame(loop);
   154→      const rawDt = Math.min(0.05, (t - this.lastFrame) / 1000);
   155→      this.lastFrame = t;
   156→      if (!this.paused && !this.gameOver) {
   157→        const dt = rawDt * this.speed;
   158→        this.update(dt);
   159→      }
   160→      this.cameraCtrl.update(rawDt);
   161→      this.scene3d.update(rawDt);
   162→      this.vfx.update(this.paused ? rawDt * 0.2 : rawDt * this.speed);
   163→      this.scene3d.render(this.cameraCtrl.camera);
   164→    };
   165→    this.raf = requestAnimationFrame(loop);
   166→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   167→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: 0 });
   168→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   169→    this.bus.emit('magic', this.magicState());
   170→    this.bus.emit('hero', this.heroState());
   171→  }

   173→  private update(dt: number) {
   174→    this.time += dt;
   175→
   176→    // موج — شمارش معکوس شروع خودکار
   177→    if (!this.waveActive && !this.gameOver) {
   178→      this.autoStartTimer -= dt;
   179→      if (this.autoStartTimer <= 0 && this.waveIndex < this.level.waves.length) {
   180→        if (this.waveIndex > 0 || !this.save.tutorialDone) {
   181→          // در آموزش شروع خودکار نداریم
   182→          if (this.save.tutorialDone) this.startWave();
   183→        } else {
   184→          this.startWave();
   185→        }
   186→      }
   187→      this.uiTimer -= dt;
   188→      if (this.uiTimer <= 0) {
   189→        this.uiTimer = 0.25;
   190→        this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: Math.max(0, this.autoStartTimer), earlyBonus: this.earlyBonus });
   191→      }
   192→    }
   193→
   194→    // اسپاون
   195→    if (this.waveActive) {
   196→      this.waveTime += dt;
   197→      while (this.spawnQueue.length && this.spawnQueue[0].time <= this.waveTime) {
   198→        const item = this.spawnQueue.shift()!;
   199→        this.spawnEnemy(item.enemyId, item.tier);
   200→      }
   201→    }
   202→
   203→    // دشمنان
   204→    for (let i = this.enemies.length - 1; i >= 0; i--) {
   205→      const e = this.enemies[i];
   206→      e.update(dt, this.time);
   207→      if (!e.alive && e.getPos().y < -2.4) {
   208→        e.remove();
   209→        this.enemies.splice(i, 1);
   210→      }
   211→    }
   212→
   213→    // اتمام موج
   214→    if (this.waveActive && this.spawnQueue.length === 0 && this.enemies.length === 0 && !this.gameOver) {
   215→      this.completeWave();
   216→    }
   217→
   218→    // برج‌ها
   219→    for (const t of this.towers) {
   220→      t.update(dt, this.time, this.enemies);
   221→    }
   222→
   223→    // پرتابه‌ها
   224→    for (let i = this.projectiles.length - 1; i >= 0; i--) {
   225→      const p = this.projectiles[i];
   226→      p.update(dt, this.scene3d.scene);
   227→      if (!p.isAlive) this.projectiles.splice(i, 1);
   228→    }
   229→
   230→    // قهرمان
   231→    this.hero.update(dt, this.time);
   232→    this.ctx.towerRateBuff = this.hero.abilityActiveTime > 0 ? HEROES.arman.abilityBuff / 100 : 0;
   233→
   234→    // رگبار آتش
   235→    this.barrageTimer -= dt;
   236→    for (let i = this.fireBarrage.length - 1; i >= 0; i--) {
   237→      const item = this.fireBarrage[i];
   238→      item.time -= dt;
   239→      if (item.time <= 0) {
   240→        const from = item.pos.clone().add(new THREE.Vector3(0, 22, 0));
   241→        this.projectiles.push(new Projectile('fireball', from, null, { damage: item.damage, splash: 2.6, burn: 4, color: 0xf97316 }, this.ctx, this.scene3d.scene));
   242→        this.fireBarrage.splice(i, 1);
   243→      }
   244→    }
   245→
   246→    // سردست‌های جادو
   247→    for (const id of Object.keys(this.magicCd)) {
   248→      this.magicCd[id] = Math.max(0, this.magicCd[id] - dt);
   249→    }
   250→
   251→    // سپر هسته
   252→    if (this.shieldUntil > 0 && this.time > this.shieldUntil) {
   253→      this.shieldUntil = 0;
   254→      this.bus.emit('shield', { active: false });
   255→    }
   256→
   257→    // بروزرسانی UI با نرخ محدود
   258→    this.uiTimer -= dt;
   259→    if (this.uiTimer <= 0) {
   260→      this.uiTimer = 0.2;
   261→      this.bus.emit('magic', this.magicState());
   262→      this.bus.emit('hero', this.heroState());
   263→      if (this.selectedTower) this.emitTowerInfo();
   264→    }
   265→  }

   267→  // ===================== موج‌ها =====================
   268→
   269→  startWave() {
   270→    if (this.waveActive || this.waveIndex >= this.level.waves.length || this.gameOver) return;
   271→    const wave = this.level.waves[this.waveIndex];
   272→    this.waveActive = true;
   273→    this.waveTime = 0;
   274→    this.spawnQueue = [];
   275→    for (const g of wave.groups) {
   276→      for (let i = 0; i < g.count; i++) {
   277→        this.spawnQueue.push({ enemyId: g.enemy, tier: g.tier, time: g.delay + i * g.interval });
   278→      }
   279→    }
   280→    this.spawnQueue.sort((a, b) => a.time - b.time);
   281→    this.earlyBonus = Math.round(15 + this.autoStartTimer * 4);
   282→    if (this.earlyBonus > 0 && this.autoStartTimer > 1) {
   283→      this.coins += this.earlyBonus;
   284→      this.coinsEarned += this.earlyBonus;
   285→      this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   286→    }
   287→    this.autoStartTimer = 0;
   288→
   289→    this.audio.play('waveStart');
   290→    this.bus.emit('banner', { wave: this.waveIndex + 1, boss: !!wave.boss });
   291→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: true, autoIn: 0 });
   292→
   293→    if (wave.boss) {
   294→      this.audio.setBossMode(true);
   295→      this.audio.play('bossRoar');
   296→      this.cameraCtrl.shake(0.6);
   297→      this.towersBuiltBeforeBoss = this.towersBuilt;
   298→    }
   299→
   300→    // پاداش کریستال موج
   301→    this.crystals += 5;
   302→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   303→
   304→    // آموزش
   305→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'upgrade');
   306→  }
   307→
   308→  private completeWave() {
   309→    const wave = this.level.waves[this.waveIndex];
   310→    this.waveActive = false;
   311→    this.waveIndex++;
   312→    this.save.stats.wavesSurvived = this.totalWavesSurvivedAtStart + this.waveIndex;
   313→    this.coins += wave.coinBonus;
   314→    this.coinsEarned += wave.coinBonus;
   315→    this.autoStartTimer = wave.boss ? 0 : 14;
   316→    this.earlyBonus = 0;
   317→    this.audio.play('reward');
   318→    this.bus.emit('waveCleared', { wave: this.waveIndex, bonus: wave.coinBonus });
   319→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   320→
   321→    if (this.waveIndex >= this.level.waves.length) {
   322→      this.doVictory();
   323→    } else {
   324→      this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: this.autoStartTimer, earlyBonus: 0 });
   325→      if (!this.save.tutorialDone) this.bus.emit('tutorial', 'magic');
   326→    }
   327→  }

   331→  private spawnEnemy(enemyId: string, tier: number) {
   332→    const data = ENEMIES[enemyId];
   333→    if (!data) return;
   334→    const enemy = new Enemy(data, tier, this.scene3d.scene, this.ctx);
   335→    this.enemies.push(enemy);
   336→    this.vfx.burst(enemy.getPos().clone().add(new THREE.Vector3(0, 1.5, 0)), data.accent, 10, { speed: 4, life: 0.5, size: 0.5, gravity: 3 });
   337→    if (data.boss) {
   338→      this.audio.play('bossRoar');
   339→      this.cameraCtrl.shake(0.7);
   340→    }
   341→  }

   343→  private summonShadow(pos: THREE.Vector3) {
   344→    const data = ENEMIES.shadow;
   345→    const e = new Enemy(data, 1, this.scene3d.scene, this.ctx);
   346→    // قرار دادن روی نزدیک‌ترین نقطه مسیر
   347→    let best = 0;
   348→    let bestD = Infinity;
   349→    this.scene3d.pathPoints.forEach((p, i) => {
   350→      const d = (p.x - pos.x) ** 2 + (p.z - pos.z) ** 2;
   351→      if (d < bestD) {
   352→        bestD = d;
   353→        best = i;
   354→      }
   355→    });
   356→    e.dist = (best / (this.scene3d.pathPoints.length - 1)) * this.scene3d.pathLength;
   357→    this.enemies.push(e);
   358→    this.vfx.burst(pos.clone().add(new THREE.Vector3(0, 1, 0)), 0xa855f7, 14, { speed: 5, life: 0.6, size: 0.6, gravity: 3 });
   359→  }
   360→
   361→  private handleKill(e: Enemy) {
   362→    this.coins += e.reward;
   363→    this.coinsEarned += e.reward;
   364→    this.crystals += e.data.crystalReward ?? 0;
   365→    this.killsThisLevel++;
   366→    this.save.stats.kills++;
   367→    if (e.data.boss) {
   368→      this.save.stats.bossKills++;
   369→      this.audio.setBossMode(false);
   370→    }
   371→    this.hero.gainXp(e.data.boss ? 200 : 12);
   372→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   373→  }
   374→
   375→  private handleLeak(e: Enemy) {
   376→    // حذف از صحنه
   377→    e.alive = false;
   378→    e.remove();
   379→    const i = this.enemies.indexOf(e);
   380→    if (i >= 0) this.enemies.splice(i, 1);
   381→
   382→    if (this.time < this.shieldUntil) {
   383→      // سپر آریانا — جلوگیری کامل
   384→      this.vfx.ring(this.scene3d.corePos, 0xfbbf24, 6, 0.5);
   385→      return;
   386→    }
   387→    this.leaks++;
   388→    this.coreHp = Math.max(0, this.coreHp - e.data.coreDamage);
   389→    this.cameraCtrl.shake(e.data.boss ? 1 : 0.45);
   390→    this.vfx.explosion(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 2, 0)), 0xef4444, e.data.boss);
   391→    this.audio.play('explosion');
   392→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   393→    if (this.coreHp <= 0 && !this.gameOver) {
   394→      this.doDefeat();
   395→    }
   396→  }

   400→  setBuildMode(towerId: string | null) {

   421→    this.ghostMesh = null;

   424→  tryBuild(towerId: string, point: THREE.Vector3): boolean {
   425→    const data = TOWERS[towerId];
   426→    if (!data) return false;
   427→    const x = Math.round(point.x / 2) * 2;
   428→    const z = Math.round(point.z / 2) * 2;
   429→    if (!this.scene3d.canBuild(x, z, this.towers.map((t) => t.position))) {
   430→      this.audio.play('error');
   431→      this.bus.emit('toast', { key: 'invalidSpot', type: 'error' });
   432→      return false;
   433→    }
   434→    if (this.coins < data.cost) {
   435→      this.audio.play('error');
   436→      this.bus.emit('toast', { key: 'notEnoughCoins', type: 'error' });
   437→      return false;
   438→    }
   439→    this.coins -= data.cost;
   440→    const tower = new Tower(this.towerNextId++, towerId, new THREE.Vector3(x, 0, z), this.scene3d.scene, this.ctx);
   441→    this.towers.push(tower);
   442→    this.towersBuilt++;
   443→    this.save.stats.towersBuilt++;
   444→    this.audio.play('place');
   445→    this.vfx.ring(tower.position, data.color, 4, 0.5);
   446→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   447→    this.bus.emit('buildMode', { towerId: null });
   448→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'wave');
   449→    return true;
   450→  }

   452→  upgradeTower(towerId: number): boolean {
   453→    const t = this.towers.find((x) => x.id === towerId);
   454→    if (!t) return false;
   455→    const cost = t.upgradeCost;
   456→    if (cost < 0) return false;
   457→    if (this.coins < cost) {
   458→      this.audio.play('error');
   459→      this.bus.emit('toast', { key: 'notEnoughCoins', type: 'error' });
   460→      return false;
   461→    }
   462→    this.coins -= cost;
   463→    t.upgrade();
   464→    this.upgradesCount++;
   465→    this.save.stats.upgrades++;
   466→    this.audio.play('upgrade');
   467→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   468→    this.emitTowerInfo();
   469→    if (t.level === 5) this.bus.emit('toast', { key: 'achMaxed', type: 'success' });
   470→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'magic');
   471→    return true;
   472→  }

   474→  sellTower(towerId: number): boolean {
   475→    const i = this.towers.findIndex((x) => x.id === towerId);
   476→    if (i < 0) return false;
   477→    const v = this.towers[i].sell();
   478→    this.towers.splice(i, 1);
   479→    this.coins += v;
   480→    this.audio.play('sell');
   481→    this.selectedTower = null;
   482→    this.bus.emit('selectTower', null);
   483→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   484→    return true;
   485→  }

   487→  useTowerAbility(towerId: number): boolean {

   489→    if (!t) return false;
   490→    return t.useAbility(this.enemies);

   493→  cycleTargeting(towerId: number) {
   494→    const t = this.towers.find((x) => x.id === towerId);

   532→        return;

   535→      if (!m.targeted) {
   536→        this.castMagic(id, this.scene3d.corePos.clone());
   537→        return;
   538→      }
   539→    }
   540→    this.magicPending = id;
   541→    this.buildMode = null;
   542→    this.heroSelectMode = false;
   543→    this.ghostMesh?.removeFromParent();
   544→    this.ghostMesh = null;

   548→  private castMagic(id: MagicId, point: THREE.Vector3): boolean {
   549→    const m = MAGIC[id];
   550→    if (!m) return false;
   551→    if (this.magicCd[id] > 0 || this.crystals < m.crystalCost) return false;
   552→
   553→    this.crystals -= m.crystalCost;
   554→    this.magicCd[id] = m.cooldown;
   555→    this.magicCastThisLevel++;
   556→    this.save.stats.magicCasts++;
   557→    this.audio.play('magic');
   558→    this.vfx.magicCircle(point, 0xf59e0b, m.radius * 1.2, 1.6);
   559→
   560→    switch (id) {
   561→      case 'skyLightning': {
   562→        const from = point.clone().add(new THREE.Vector3(0, 34, 0));
   563→        this.vfx.bolt(from, point.clone(), 0xc4b5fd, true);
   564→        this.vfx.bolt(from.clone().add(new THREE.Vector3(2, 0, 1)), point.clone().add(new THREE.Vector3(0.5, 0, -0.5)), 0xa78bfa, true);
   565→        this.vfx.explosion(point, 0xa78bfa, true);
   566→        this.cameraCtrl.shake(0.55);
   567→        this.audio.play('lightning');
   568→        // آسیب ناحیه‌ای
   569→        for (const e of [...this.enemies]) {
   570→          if (!e.alive) continue;
   571→          const d = e.getPos().distanceTo(point);
   572→          if (d <= m.radius) {
   573→            e.takeDamage(m.damage! * (1 - (d / m.radius) * 0.4), 'energy');
   574→          }
   575→        }
   576→        break;
   577→      }
   578→      case 'fireRain': {
   579→        for (let i = 0; i < 12; i++) {
   580→          const a = Math.random() * Math.PI * 2;
   581→          const r = Math.random() * m.radius;
   582→          this.fireBarrage.push({
   583→            time: (i / 12) * (m.duration ?? 4),
   584→            pos: point.clone().add(new THREE.Vector3(Math.cos(a) * r, 0, Math.sin(a) * r)),
   585→            damage: m.damage ?? 38,
   586→          });
   587→        }
   588→        this.audio.play('cannon');
   589→        break;
   590→      }
   591→      case 'arianaShield': {
   592→        this.shieldUntil = this.time + (m.duration ?? 10);
   593→        this.vfx.ring(this.scene3d.corePos, 0xfbbf24, 6, 0.9);
   594→        this.vfx.burst(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 3, 0)), 0xfbbf24, 40, { speed: 8, life: 1, size: 0.7, gravity: 1, up: 1 });
   595→        this.bus.emit('shield', { active: true, duration: m.duration });
   596→        break;
   597→      }
   598→    }
   599→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   600→    this.bus.emit('magic', this.magicState());
   601→    if (!this.save.tutorialDone) this.bus.emit('tutorial', { step: 'done' });
   602→    return true;
   603→  }

   629→  private chainLightning(from: THREE.Vector3, first: DamageableEnemy, chains: number, damage: number, color: number) {
   630→    const hitList: DamageableEnemy[] = [];
   631→    let current = first;
   632→    let currentDamage = damage;
   633→    let currentFrom = from;
   634→    for (let i = 0; i <= chains; i++) {
   635→      if (!current || !current.alive) break;
   636→      this.vfx.bolt(currentFrom, current.getPos().clone().add(new THREE.Vector3(0, 1.2, 0)), color);
   637→      hitList.push(current);
   638→      current.takeDamage(currentDamage, 'energy');
   639→      currentDamage *= 0.72;
   640→      currentFrom = current.getPos().clone().add(new THREE.Vector3(0, 1, 0));
   641→      // هدف بعدی
   642→      let next: DamageableEnemy | null = null;
   643→      let bestD = 8;
   644→      for (const e of this.enemies) {
   645→        if (!e.alive || hitList.includes(e)) continue;
   646→        const d = e.getPos().distanceTo(currentFrom);
   647→        if (d < bestD) {
   648→          bestD = d;
   649→          next = e;
   650→        }
   651→      }
   652→      current = next as DamageableEnemy;
   653→      if (!next) break;
   654→    }
   655→  }

   657→  private scheduleFireBarrage(center: THREE.Vector3, radius: number, count: number, damage: number) {
   658→    for (let i = 0; i < count; i++) {

   660→      const r = Math.random() * radius;
   661→      this.fireBarrage.push({
   662→        time: i * 0.3,
   663→        pos: center.clone().add(new THREE.Vector3(Math.cos(a) * r, 0, Math.sin(a) * r)),
   664→        damage,
   665→      });
   666→    }

   671→  private handleTap(hit: TapHit) {
   672→    if (this.gameOver) return;
   673→
   674→    // انتخاب شیء؟
   675→    if (hit.objects.length > 0) {
   676→      const root = hit.objects[0].object;
   677→      // برج؟
   678→      const tower = this.towers.find((t) => {
   679→        let o: THREE.Object3D | null = root;
   680→        while (o) {
   681→          if (o === t.mesh) return true;
   682→          o = o.parent;
   683→        }
   684→        return false;
   685→      });
   686→      if (tower) {
   687→        this.selectedTower = tower;
   688→        this.emitTowerInfo();
   689→        this.buildMode = null;
   690→        this.heroSelectMode = false;
   691→        this.ghostMesh?.removeFromParent();
   692→        this.ghostMesh = null;
   693→        this.audio.play('tap');
   694→        return;
   695→      }
   696→      // قهرمان؟
   697→      if (root === this.hero.mesh || (root.parent && root.parent === this.hero.mesh)) {
   698→        this.bus.emit('heroSelected', {});
   699→        this.audio.play('tap');
   700→        return;
   701→      }
   702→    }
   703→
   704→    // جادو در انتظار هدف
   705→    if (this.magicPending) {
   706→      const id = this.magicPending;
   707→      this.magicPending = null;
   708→      this.heroSelectMode = false;
   709→      this.bus.emit('magicPending', { id: null });
   710→      this.castMagic(id, hit.ground);
   711→      return;
   712→    }
   713→
   714→    // حالت ساخت
   715→    if (this.buildMode) {
   716→      this.tryBuild(this.buildMode, hit.ground);
   717→      this.vfx.ring(hit.ground, 0xffffff, 2, 0.3);
   718→      return;
   719→    }
   720→
   721→    // حرکت/استقرار قهرمان
   722→    if (this.heroSelectMode) {
   723→      if (!this.hero.deployed) {
   724→        this.deployHero(hit.ground);
   725→      } else {
   726→        this.hero.moveTo(hit.ground);
   727→        this.vfx.ring(hit.ground, 0xffb020, 1.5, 0.4);
   728→      }
   729→      this.selectHeroMode(false);
   730→      return;
   731→    }
   732→
   733→    this.selectedTower = null;
   734→    this.bus.emit('selectTower', null);
   735→  }

   753→  private doVictory() {
   754→    this.gameOver = true;
   755→    const healthPct = (this.coreHp / this.coreMaxHp) * 100;
   756→    const star1 = true;
   757→    const star2 = healthPct >= this.level.star2HealthPct;
   758→    const star3 = this.leaks === 0;
   759→    const stars = (star1 ? 1 : 0) + (star2 ? 1 : 0) + (star3 ? 1 : 0);
   760→
   761→    // پاداش‌ها
   762→    const rewardCrystals = 10 + stars * 5;
   763→    const rewardKnowledge = 2 + stars;
   764→    this.crystals += rewardCrystals;
   765→    this.save.crystals = this.crystals;
   766→    this.save.knowledge += rewardKnowledge;
   767→    this.save.stats.victories++;
   768→    if (this.leaks === 0 && !this.save.achievements.includes('untouchable')) this.save.achievements.push('untouchable');
   769→    if (this.coreHp === this.coreMaxHp && !this.save.achievements.includes('perfect_defense')) this.save.achievements.push('perfect_defense');
   770→    if (healthPct < 10 && !this.save.achievements.includes('last_stand')) this.save.achievements.push('last_stand');
   771→
   772→    const prev = this.save.levels[this.level.id];
   773→    const bestStars = Math.max(prev?.stars ?? 0, stars);
   774→    this.save.levels[this.level.id] = {
   775→      stars: bestStars,
   776→      bestHealthPct: Math.max(prev?.bestHealthPct ?? 0, Math.round(healthPct)),
   777→      completed: true,
   778→    };
   779→    // بازمحاسبه مجموع ستاره‌ها
   780→    this.save.stats.totalStars = Object.values(this.save.levels).reduce((sum, l) => sum + l.stars, 0);
   781→    this.save.tutorialDone = true;
   782→
   783→    this.audio.play('victory');
   784→    this.vfx.burst(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 3, 0)), 0xfbbf24, 80, { speed: 10, life: 1.4, size: 0.8, gravity: 4, up: 1 });
   785→    this.bus.emit('victory', {
   786→      stars,
   787→      healthPct: Math.round(healthPct),
   788→      rewards: { crystals: rewardCrystals, knowledge: rewardKnowledge, coins: this.coinsEarned },
   789→    });
   790→  }
