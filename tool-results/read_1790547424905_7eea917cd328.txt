   148→   148→  start() {
   149→   149→    this.lastFrame = performance.now();
   150→   150→    const loop = (t: number) => {
   151→   151→      if (this.disposed) return;
   152→   152→      this.raf = requestAnimationFrame(loop);
   153→   153→      const rawDt = Math.min(0.05, (t - this.lastFrame) / 1000);
   154→   154→      this.lastFrame = t;
   155→   155→      if (!this.paused && !this.gameOver) {
   156→   156→        const dt = rawDt * this.speed;
   157→   157→        this.update(dt);
   158→   158→      }
   159→   159→      this.cameraCtrl.update(rawDt);
   160→   160→      this.scene3d.update(rawDt);
   161→   161→      this.vfx.update(this.paused ? rawDt * 0.2 : rawDt * this.speed);
   162→   162→      this.scene3d.render(this.cameraCtrl.camera);
   163→   163→    };
   164→   164→    this.raf = requestAnimationFrame(loop);
   165→   165→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   166→   166→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: 0 });
   167→   167→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   168→   168→    this.bus.emit('magic', this.magicState());
   169→   169→    this.bus.emit('hero', this.heroState());
   170→   170→  }

   172→   172→  private update(dt: number) {
   173→   173→    this.time += dt;
   174→   174→
   175→   175→    // موج — شمارش معکوس شروع خودکار
   176→   176→    if (!this.waveActive && !this.gameOver) {
   177→   177→      this.autoStartTimer -= dt;
   178→   178→      if (this.autoStartTimer <= 0 && this.waveIndex < this.level.waves.length) {
   179→   179→        if (this.waveIndex > 0 || !this.save.tutorialDone) {
   180→   180→          // در آموزش شروع خودکار نداریم
   181→   181→          if (this.save.tutorialDone) this.startWave();
   182→   182→        } else {
   183→   183→          this.startWave();
   184→   184→        }
   185→   185→      }
   186→   186→      this.uiTimer -= dt;
   187→   187→      if (this.uiTimer <= 0) {
   188→   188→        this.uiTimer = 0.25;
   189→   189→        this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: Math.max(0, this.autoStartTimer), earlyBonus: this.earlyBonus });
   190→   190→      }
   191→   191→    }
   192→   192→
   193→   193→    // اسپاون
   194→   194→    if (this.waveActive) {
   195→   195→      this.waveTime += dt;
   196→   196→      while (this.spawnQueue.length && this.spawnQueue[0].time <= this.waveTime) {
   197→   197→        const item = this.spawnQueue.shift()!;
   198→   198→        this.spawnEnemy(item.enemyId, item.tier);
   199→   199→      }
   200→   200→    }
   201→   201→
   202→   202→    // دشمنان
   203→   203→    for (let i = this.enemies.length - 1; i >= 0; i--) {
   204→   204→      const e = this.enemies[i];
   205→   205→      e.update(dt, this.time);
   206→   206→      if (!e.alive && e.getPos().y < -2.4) {
   207→   207→        e.remove();
   208→   208→        this.enemies.splice(i, 1);
   209→   209→      }
   210→   210→    }
   211→   211→
   212→   212→    // اتمام موج
   213→   213→    if (this.waveActive && this.spawnQueue.length === 0 && this.enemies.length === 0 && !this.gameOver) {
   214→   214→      this.completeWave();
   215→   215→    }
   216→   216→
   217→   217→    // برج‌ها
   218→   218→    for (const t of this.towers) {
   219→   219→      t.update(dt, this.time, this.enemies);
   220→   220→    }
   221→   221→
   222→   222→    // پرتابه‌ها
   223→   223→    for (let i = this.projectiles.length - 1; i >= 0; i--) {
   224→   224→      const p = this.projectiles[i];
   225→   225→      p.update(dt, this.scene3d.scene);
   226→   226→      if (!p.isAlive) this.projectiles.splice(i, 1);
   227→   227→    }
   228→   228→
   229→   229→    // قهرمان
   230→   230→    this.hero.update(dt, this.time);
   231→   231→    this.ctx.towerRateBuff = this.hero.abilityActiveTime > 0 ? HEROES.arman.abilityBuff / 100 : 0;
   232→   232→
   233→   233→    // رگبار آتش
   234→   234→    this.barrageTimer -= dt;
   235→   235→    for (let i = this.fireBarrage.length - 1; i >= 0; i--) {
   236→   236→      const item = this.fireBarrage[i];
   237→   237→      item.time -= dt;
   238→   238→      if (item.time <= 0) {
   239→   239→        const from = item.pos.clone().add(new THREE.Vector3(0, 22, 0));
   240→   240→        this.projectiles.push(new Projectile('fireball', from, null, { damage: item.damage, splash: 2.6, burn: 4, color: 0xf97316 }, this.ctx, this.scene3d.scene));
   241→   241→        this.fireBarrage.splice(i, 1);
   242→   242→      }
   243→   243→    }
   244→   244→
   245→   245→    // سردست‌های جادو
   246→   246→    for (const id of Object.keys(this.magicCd)) {
   247→   247→      this.magicCd[id] = Math.max(0, this.magicCd[id] - dt);
   248→   248→    }
   249→   249→
   250→   250→    // سپر هسته
   251→   251→    if (this.shieldUntil > 0 && this.time > this.shieldUntil) {
   252→   252→      this.shieldUntil = 0;
   253→   253→      this.bus.emit('shield', { active: false });
   254→   254→    }
   255→   255→
   256→   256→    // بروزرسانی UI با نرخ محدود
   257→   257→    this.uiTimer -= dt;
   258→   258→    if (this.uiTimer <= 0) {
   259→   259→      this.uiTimer = 0.2;
   260→   260→      this.bus.emit('magic', this.magicState());
   261→   261→      this.bus.emit('hero', this.heroState());
   262→   262→      if (this.selectedTower) this.emitTowerInfo();
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  // ===================== موج‌ها =====================
   267→   267→
   268→   268→  startWave() {
   269→   269→    if (this.waveActive || this.waveIndex >= this.level.waves.length || this.gameOver) return;
   270→   270→    const wave = this.level.waves[this.waveIndex];
   271→   271→    this.waveActive = true;
   272→   272→    this.waveTime = 0;
   273→   273→    this.spawnQueue = [];
   274→   274→    for (const g of wave.groups) {
   275→   275→      for (let i = 0; i < g.count; i++) {
   276→   276→        this.spawnQueue.push({ enemyId: g.enemy, tier: g.tier, time: g.delay + i * g.interval });
   277→   277→      }
   278→   278→    }
   279→   279→    this.spawnQueue.sort((a, b) => a.time - b.time);
   280→   280→    this.earlyBonus = Math.round(15 + this.autoStartTimer * 4);
   281→   281→    if (this.earlyBonus > 0 && this.autoStartTimer > 1) {
   282→   282→      this.coins += this.earlyBonus;
   283→   283→      this.coinsEarned += this.earlyBonus;
   284→   284→      this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   285→   285→    }
   286→   286→    this.autoStartTimer = 0;
   287→   287→
   288→   288→    this.audio.play('waveStart');
   289→   289→    this.bus.emit('banner', { wave: this.waveIndex + 1, boss: !!wave.boss });
   290→   290→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: true, autoIn: 0 });
   291→   291→
   292→   292→    if (wave.boss) {
   293→   293→      this.audio.setBossMode(true);
   294→   294→      this.audio.play('bossRoar');
   295→   295→      this.cameraCtrl.shake(0.6);
   296→   296→      this.towersBuiltBeforeBoss = this.towersBuilt;
   297→   297→    }
   298→   298→
   299→   299→    // پاداش کریستال موج
   300→   300→    this.crystals += 5;
   301→   301→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   302→   302→
   303→   303→    // آموزش
   304→   304→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'upgrade');
   305→   305→  }
   306→   306→
   307→   307→  private completeWave() {
   308→   308→    const wave = this.level.waves[this.waveIndex];
   309→   309→    this.waveActive = false;
   310→   310→    this.waveIndex++;
   311→   311→    this.save.stats.wavesSurvived = this.totalWavesSurvivedAtStart + this.waveIndex;
   312→   312→    this.coins += wave.coinBonus;
   313→   313→    this.coinsEarned += wave.coinBonus;
   314→   314→    this.autoStartTimer = wave.boss ? 0 : 14;
   315→   315→    this.earlyBonus = 0;
   316→   316→    this.audio.play('reward');
   317→   317→    this.bus.emit('waveCleared', { wave: this.waveIndex, bonus: wave.coinBonus });
   318→   318→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   319→   319→
   320→   320→    if (this.waveIndex >= this.level.waves.length) {
   321→   321→      this.doVictory();
   322→   322→    } else {
   323→   323→      this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: this.autoStartTimer, earlyBonus: 0 });
   324→   324→      if (!this.save.tutorialDone) this.bus.emit('tutorial', 'magic');
   325→   325→    }
   326→   326→  }

   330→   330→  private spawnEnemy(enemyId: string, tier: number) {
   331→   331→    const data = ENEMIES[enemyId];
   332→   332→    if (!data) return;
   333→   333→    const enemy = new Enemy(data, tier, this.scene3d.scene, this.ctx);
   334→   334→    this.enemies.push(enemy);
   335→   335→    this.vfx.burst(enemy.getPos().clone().add(new THREE.Vector3(0, 1.5, 0)), data.accent, 10, { speed: 4, life: 0.5, size: 0.5, gravity: 3 });
   336→   336→    if (data.boss) {
   337→   337→      this.audio.play('bossRoar');
   338→   338→      this.cameraCtrl.shake(0.7);
   339→   339→    }
   340→   340→  }

   342→   342→  private summonShadow(pos: THREE.Vector3) {
   343→   343→    const data = ENEMIES.shadow;
   344→   344→    const e = new Enemy(data, 1, this.scene3d.scene, this.ctx);
   345→   345→    // قرار دادن روی نزدیک‌ترین نقطه مسیر
   346→   346→    let best = 0;
   347→   347→    let bestD = Infinity;
   348→   348→    this.scene3d.pathPoints.forEach((p, i) => {
   349→   349→      const d = (p.x - pos.x) ** 2 + (p.z - pos.z) ** 2;
   350→   350→      if (d < bestD) {
   351→   351→        bestD = d;
   352→   352→        best = i;
   353→   353→      }
   354→   354→    });
   355→   355→    e.dist = (best / (this.scene3d.pathPoints.length - 1)) * this.scene3d.pathLength;
   356→   356→    this.enemies.push(e);
   357→   357→    this.vfx.burst(pos.clone().add(new THREE.Vector3(0, 1, 0)), 0xa855f7, 14, { speed: 5, life: 0.6, size: 0.6, gravity: 3 });
   358→   358→  }
   359→   359→
   360→   360→  private handleKill(e: Enemy) {
   361→   361→    this.coins += e.reward;
   362→   362→    this.coinsEarned += e.reward;
   363→   363→    this.crystals += e.data.crystalReward ?? 0;
   364→   364→    this.killsThisLevel++;
   365→   365→    this.save.stats.kills++;
   366→   366→    if (e.data.boss) {
   367→   367→      this.save.stats.bossKills++;
   368→   368→      this.audio.setBossMode(false);
   369→   369→    }
   370→   370→    this.hero.gainXp(e.data.boss ? 200 : 12);
   371→   371→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   372→   372→  }
   373→   373→
   374→   374→  private handleLeak(e: Enemy) {
   375→   375→    // حذف از صحنه
   376→   376→    e.alive = false;
   377→   377→    e.remove();
   378→   378→    const i = this.enemies.indexOf(e);
   379→   379→    if (i >= 0) this.enemies.splice(i, 1);
   380→   380→
   381→   381→    if (this.time < this.shieldUntil) {
   382→   382→      // سپر آریانا — جلوگیری کامل
   383→   383→      this.vfx.ring(this.scene3d.corePos, 0xfbbf24, 6, 0.5);
   384→   384→      return;
   385→   385→    }
   386→   386→    this.leaks++;
   387→   387→    this.coreHp = Math.max(0, this.coreHp - e.data.coreDamage);
   388→   388→    this.cameraCtrl.shake(e.data.boss ? 1 : 0.45);
   389→   389→    this.vfx.explosion(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 2, 0)), 0xef4444, e.data.boss);
   390→   390→    this.audio.play('explosion');
   391→   391→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   392→   392→    if (this.coreHp <= 0 && !this.gameOver) {
   393→   393→      this.doDefeat();
   394→   394→    }
   395→   395→  }

   415→   415→  selectHeroMode(on: boolean) {
   416→   416→    this.heroSelectMode = on;

   547→   547→  private castMagic(id: MagicId, point: THREE.Vector3): boolean {
   548→   548→    const m = MAGIC[id];
   549→   549→    if (!m) return false;
   550→   550→    if (this.magicCd[id] > 0 || this.crystals < m.crystalCost) return false;
   551→   551→

   553→   553→    this.magicCd[id] = m.cooldown;

   555→   555→    this.save.stats.magicCasts++;
   556→   556→    this.audio.play('magic');
   557→   557→    this.vfx.magicCircle(point, 0xf59e0b, m.radius * 1.2, 1.6);
   558→   558→
   559→   559→    switch (id) {
   560→   560→      case 'skyLightning': {
   561→   561→        const from = point.clone().add(new THREE.Vector3(0, 34, 0));
   562→   562→        this.vfx.bolt(from, point.clone(), 0xc4b5fd, true);
   563→   563→        this.vfx.bolt(from.clone().add(new THREE.Vector3(2, 0, 1)), point.clone().add(new THREE.Vector3(0.5, 0, -0.5)), 0xa78bfa, true);

   569→   569→          if (!e.alive) continue;
   570→   570→          const d = e.getPos().distanceTo(point);
   571→   571→          if (d <= m.radius) {
   572→   572→            e.takeDamage(m.damage! * (1 - (d / m.radius) * 0.4), 'energy');
   573→   573→          }
   574→   574→        }
   575→   575→        break;
   576→   576→      }
   577→   577→      case 'fireRain': {

   580→   580→          const r = Math.random() * m.radius;
   581→   581→          this.fireBarrage.push({
   582→   582→            time: (i / 12) * (m.duration ?? 4),
   583→   583→            pos: point.clone().add(new THREE.Vector3(Math.cos(a) * r, 0, Math.sin(a) * r)),
   584→   584→            damage: m.damage ?? 38,
   585→   585→          });

   588→   588→        break;

   594→   594→        this.bus.emit('shield', { active: true, duration: m.duration });
   595→   595→        break;
   596→   596→      }
   597→   597→    }
   598→   598→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   599→   599→    this.bus.emit('magic', this.magicState());
   600→   600→    if (!this.save.tutorialDone) this.bus.emit('tutorial', { step: 'done' });

   602→   602→  }

   628→   628→  private chainLightning(from: THREE.Vector3, first: DamageableEnemy, chains: number, damage: number, color: number) {
   629→   629→    const hitList: DamageableEnemy[] = [];
   630→   630→    let current = first;
   631→   631→    let currentDamage = damage;
   632→   632→    let currentFrom = from;
   633→   633→    for (let i = 0; i <= chains; i++) {
   634→   634→      if (!current || !current.alive) break;
   635→   635→      this.vfx.bolt(currentFrom, current.getPos().clone().add(new THREE.Vector3(0, 1.2, 0)), color);
   636→   636→      hitList.push(current);
   637→   637→      current.takeDamage(currentDamage, 'energy');
   638→   638→      currentDamage *= 0.72;
   639→   639→      currentFrom = current.getPos().clone().add(new THREE.Vector3(0, 1, 0));
   640→   640→      // هدف بعدی
   641→   641→      let next: DamageableEnemy | null = null;
   642→   642→      let bestD = 8;
   643→   643→      for (const e of this.enemies) {
   644→   644→        if (!e.alive || hitList.includes(e)) continue;
   645→   645→        const d = e.getPos().distanceTo(currentFrom);
   646→   646→        if (d < bestD) {
   647→   647→          bestD = d;
   648→   648→          next = e;
   649→   649→        }
   650→   650→      }
   651→   651→      current = next as DamageableEnemy;
   652→   652→      if (!next) break;
   653→   653→    }
   654→   654→  }

   670→   670→  private handleTap(hit: TapHit) {
   671→   671→    if (this.gameOver) return;
   672→   672→
   673→   673→    // انتخاب شیء؟
   674→   674→    if (hit.objects.length > 0) {
   675→   675→      const root = hit.objects[0].object;
   676→   676→      // برج؟
   677→   677→      const tower = this.towers.find((t) => {
   678→   678→        let o: THREE.Object3D | null = root;
   679→   679→        while (o) {
   680→   680→          if (o === t.mesh) return true;
   681→   681→          o = o.parent;
   682→   682→        }
   683→   683→        return false;
   684→   684→      });
   685→   685→      if (tower) {
   686→   686→        this.selectedTower = tower;
   687→   687→        this.emitTowerInfo();
   688→   688→        this.buildMode = null;
   689→   689→        this.heroSelectMode = false;
   690→   690→        this.ghostMesh?.removeFromParent();
   691→   691→        this.ghostMesh = null;
   692→   692→        this.audio.play('tap');
   693→   693→        return;
   694→   694→      }
   695→   695→      // قهرمان؟
   696→   696→      if (root === this.hero.mesh || (root.parent && root.parent === this.hero.mesh)) {
   697→   697→        this.bus.emit('heroSelected', {});
   698→   698→        this.audio.play('tap');
   699→   699→        return;
   700→   700→      }
   701→   701→    }
   702→   702→
   703→   703→    // جادو در انتظار هدف
   704→   704→    if (this.magicPending) {
   705→   705→      const id = this.magicPending;
   706→   706→      this.magicPending = null;
   707→   707→      this.heroSelectMode = false;
   708→   708→      this.bus.emit('magicPending', { id: null });
   709→   709→      this.castMagic(id, hit.ground);
   710→   710→      return;
   711→   711→    }
   712→   712→
   713→   713→    // حالت ساخت
   714→   714→    if (this.buildMode) {
   715→   715→      this.tryBuild(this.buildMode, hit.ground);
   716→   716→      this.vfx.ring(hit.ground, 0xffffff, 2, 0.3);
   717→   717→      return;
   718→   718→    }
   719→   719→
   720→   720→    // حرکت/استقرار قهرمان
   721→   721→    if (this.heroSelectMode) {
   722→   722→      if (!this.hero.deployed) {
   723→   723→        this.deployHero(hit.ground);
   724→   724→      } else {
   725→   725→        this.hero.moveTo(hit.ground);
   726→   726→        this.vfx.ring(hit.ground, 0xffb020, 1.5, 0.4);
   727→   727→      }
   728→   728→      this.selectHeroMode(false);
   729→   729→      return;
   730→   730→    }
   731→   731→
   732→   732→    this.selectedTower = null;
   733→   733→    this.bus.emit('selectTower', null);
   734→   734→  }

   774→   774→      stars: bestStars,
