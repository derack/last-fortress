   168→  start() {
   169→    this.lastFrame = performance.now();
   170→    const loop = (t: number) => {
   171→      if (this.disposed) return;
   172→      this.raf = requestAnimationFrame(loop);
   173→      const rawDt = Math.min(0.05, (t - this.lastFrame) / 1000);
   174→      this.lastFrame = t;
   175→      if (!this.paused && !this.gameOver) {
   176→        const dt = rawDt * this.speed;
   177→        this.update(dt);
   178→      }
   179→      this.cameraCtrl.update(rawDt);
   180→      this.scene3d.update(rawDt);
   181→      this.vfx.update(this.paused ? rawDt * 0.2 : rawDt * this.speed);
   182→      this.scene3d.render(this.cameraCtrl.camera);
   183→    };
   184→    this.raf = requestAnimationFrame(loop);
   185→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   186→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: 0 });
   187→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   188→    this.bus.emit('magic', this.magicState());
   189→    this.bus.emit('hero', this.heroState());
   190→  }

   192→  private update(dt: number) {
   193→    this.time += dt;
   194→
   195→    // موج — شمارش معکوس شروع خودکار
   196→    if (!this.waveActive && !this.gameOver) {
   197→      this.autoStartTimer -= dt;
   198→      if (this.autoStartTimer <= 0 && this.waveIndex < this.level.waves.length) {
   199→        if (this.waveIndex > 0 || !this.save.tutorialDone) {
   200→          // در آموزش شروع خودکار نداریم
   201→          if (this.save.tutorialDone) this.startWave();
   202→        } else {
   203→          this.startWave();
   204→        }
   205→      }
   206→      this.uiTimer -= dt;
   207→      if (this.uiTimer <= 0) {
   208→        this.uiTimer = 0.25;
   209→        this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: Math.max(0, this.autoStartTimer), earlyBonus: this.earlyBonus });
   210→      }
   211→    }
   212→
   213→    // اسپاون
   214→    if (this.waveActive) {
   215→      this.waveTime += dt;
   216→      while (this.spawnQueue.length && this.spawnQueue[0].time <= this.waveTime) {
   217→        const item = this.spawnQueue.shift()!;
   218→        this.spawnEnemy(item.enemyId, item.tier);
   219→      }
   220→    }
   221→
   222→    // دشمنان
   223→    const camPos = this.cameraCtrl.camera.position;
   224→    for (let i = this.enemies.length - 1; i >= 0; i--) {
   225→      const e = this.enemies[i];
   226→      e.update(dt, this.time);
   227→      e.updateLod(camPos);
   228→      if (!e.alive && e.getPos().y < -2.4) {
   229→        e.remove();
   230→        this.enemies.splice(i, 1);
   231→      }
   232→    }
   233→
   234→    // اتمام موج
   235→    if (this.waveActive && this.spawnQueue.length === 0 && this.enemies.length === 0 && !this.gameOver) {
   236→      this.completeWave();
   237→    }
   238→
   239→    // برج‌ها
   240→    for (const t of this.towers) {
   241→      t.update(dt, this.time, this.enemies);
   242→    }
   243→
   244→    // پرتابه‌ها
   245→    for (let i = this.projectiles.length - 1; i >= 0; i--) {
   246→      const p = this.projectiles[i];
   247→      p.update(dt, this.scene3d.scene);
   248→      if (!p.isAlive) this.projectiles.splice(i, 1);
   249→    }
   250→
   251→    // نبض پیش‌نمایش ساخت
   252→    if (this.ghostGroup && this.ghostGroup.visible) {
   253→      this.ghostPulse += dt * 4;
   254→      const pulse = 0.32 + Math.sin(this.ghostPulse) * 0.1;
   255→      const color = this.ghostValid ? 0x22c55e : 0xef4444;
   256→      (this.ghostDisc!.material as THREE.MeshBasicMaterial).opacity = pulse;
   257→      (this.ghostDisc!.material as THREE.MeshBasicMaterial).color.setHex(color);
   258→      (this.ghostRing!.material as THREE.MeshBasicMaterial).color.setHex(color);
   259→    }
   260→
   261→    // قهرمان
   262→    this.hero.update(dt, this.time);
   263→    this.hero.updateLod(this.cameraCtrl.camera.position);
   264→    // حلقه پالس حالت جابجایی — زیر قهرمان تا کاربر بداند کجا تپ کند
   265→    if (this.heroSelectMode && this.hero.alive) {
   266→      this.heroMovePulse -= dt;
   267→      if (this.heroMovePulse <= 0) {
   268→        this.heroMovePulse = 0.55;
   269→        this.vfx.ring(this.hero.getPos(), 0xffb020, 2.2, 0.5);
   270→      }
   271→    }
   272→    const commandBuff = this.hero.data.abilityKind === 'battleCommand' && this.hero.abilityActiveTime > 0
   273→      ? this.hero.data.abilityBuff / 100 : 0;
   274→    const blessingBuff = this.time < this.blessingUntil
   275→      ? (MAGIC.azarBlessing.rateBuffPct ?? 0) : 0;
   276→    this.ctx.towerRateBuff = commandBuff + blessingBuff;
   277→
   278→    // رگبار آتش/تیر
   279→    this.barrageTimer -= dt;
   280→    for (let i = this.fireBarrage.length - 1; i >= 0; i--) {
   281→      const item = this.fireBarrage[i];
   282→      item.time -= dt;
   283→      if (item.time <= 0) {
   284→        if (item.visual === 'arrow') {
   285→          const from = item.pos.clone().add(new THREE.Vector3((Math.random() - 0.5) * 8, 16, (Math.random() - 0.5) * 8));
   286→          this.projectiles.push(new Projectile('bolt', from, null, { damage: item.damage, color: 0xc084fc }, this.ctx, this.scene3d.scene));
   287→        } else {
   288→          const from = item.pos.clone().add(new THREE.Vector3(0, 22, 0));
   289→          this.projectiles.push(new Projectile('fireball', from, null, { damage: item.damage, splash: 2.6, burn: 4, color: 0xf97316 }, this.ctx, this.scene3d.scene));
   290→        }
   291→        this.fireBarrage.splice(i, 1);
   292→      }
   293→    }
   294→
   295→    // سردست‌های جادو
   296→    for (const id of Object.keys(this.magicCd)) {
   297→      this.magicCd[id] = Math.max(0, this.magicCd[id] - dt);
   298→    }
   299→
   300→    // سپر هسته
   301→    if (this.shieldUntil > 0 && this.time > this.shieldUntil) {
   302→      this.shieldUntil = 0;
   303→      this.bus.emit('shield', { active: false });
   304→    }
   305→
   306→    // بروزرسانی UI با نرخ محدود
   307→    this.uiTimer -= dt;
   308→    if (this.uiTimer <= 0) {
   309→      this.uiTimer = 0.2;
   310→      this.bus.emit('magic', this.magicState());
   311→      this.bus.emit('hero', this.heroState());
   312→      if (this.selectedTower) this.emitTowerInfo();
   313→    }
   314→  }
   315→
   316→  // ===================== موج‌ها =====================
   317→
   318→  startWave() {
   319→    if (this.waveActive || this.waveIndex >= this.level.waves.length || this.gameOver) return;
   320→    const wave = this.level.waves[this.waveIndex];
   321→    this.waveActive = true;
   322→    this.waveTime = 0;
   323→    this.spawnQueue = [];
   324→    for (const g of wave.groups) {
   325→      for (let i = 0; i < g.count; i++) {
   326→        this.spawnQueue.push({ enemyId: g.enemy, tier: g.tier, time: g.delay + i * g.interval });
   327→      }
   328→    }
   329→    this.spawnQueue.sort((a, b) => a.time - b.time);
   330→    this.earlyBonus = Math.round(15 + this.autoStartTimer * 4);
   331→    if (this.earlyBonus > 0 && this.autoStartTimer > 1) {
   332→      this.coins += this.earlyBonus;
   333→      this.coinsEarned += this.earlyBonus;
   334→      this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   335→    }
   336→    this.autoStartTimer = 0;
   337→
   338→    this.audio.play('waveStart');
   339→    this.bus.emit('banner', { wave: this.waveIndex + 1, boss: !!wave.boss });
   340→    this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: true, autoIn: 0 });
   341→
   342→    if (wave.boss) {
   343→      this.audio.setBossMode(true);
   344→      this.audio.play('bossRoar');
   345→      this.cameraCtrl.shake(0.6);
   346→      this.towersBuiltBeforeBoss = this.towersBuilt;
   347→    }
   348→
   349→    // پاداش کریستال موج
   350→    this.crystals += 5;
   351→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   352→
   353→    // آموزش
   354→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'upgrade');
   355→  }
   356→
   357→  private completeWave() {
   358→    const wave = this.level.waves[this.waveIndex];
   359→    this.waveActive = false;
   360→    this.waveIndex++;
   361→    this.save.stats.wavesSurvived = this.totalWavesSurvivedAtStart + this.waveIndex;
   362→    this.coins += wave.coinBonus;
   363→    this.coinsEarned += wave.coinBonus;
   364→    this.autoStartTimer = wave.boss ? 0 : 14;
   365→    this.earlyBonus = 0;
   366→    this.audio.play('reward');
   367→    this.bus.emit('waveCleared', { wave: this.waveIndex, bonus: wave.coinBonus });
   368→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   369→
   370→    if (this.waveIndex >= this.level.waves.length) {
   371→      this.doVictory();
   372→    } else {
   373→      this.bus.emit('wave', { wave: this.waveIndex + 1, total: this.level.waves.length, active: false, autoIn: this.autoStartTimer, earlyBonus: 0 });
   374→      if (!this.save.tutorialDone) this.bus.emit('tutorial', 'magic');
   375→    }
   376→  }

   380→  private spawnEnemy(enemyId: string, tier: number) {
   381→    const data = ENEMIES[enemyId];
   382→    if (!data) return;
   383→    const enemy = new Enemy(data, tier, this.scene3d.scene, this.ctx, stagePowerMul(this.level.region));
   384→    this.enemies.push(enemy);
   385→    this.vfx.burst(enemy.getPos().clone().add(new THREE.Vector3(0, 1.5, 0)), data.accent, 10, { speed: 4, life: 0.5, size: 0.5, gravity: 3 });
   386→    if (data.boss) {
   387→      this.audio.play('bossRoar');
   388→      this.cameraCtrl.shake(0.7);
   389→    }
   390→  }

   392→  private summonShadow(pos: THREE.Vector3, enemyId = 'shadow') {
   393→    const data = ENEMIES[enemyId] ?? ENEMIES.shadow;
   394→    const e = new Enemy(data, 1, this.scene3d.scene, this.ctx, stagePowerMul(this.level.region));
   395→    // قرار دادن روی نزدیک‌ترین نقطه مسیر
   396→    let best = 0;
   397→    let bestD = Infinity;
   398→    this.scene3d.pathPoints.forEach((p, i) => {
   399→      const d = (p.x - pos.x) ** 2 + (p.z - pos.z) ** 2;
   400→      if (d < bestD) {
   401→        bestD = d;
   402→        best = i;
   403→      }
   404→    });
   405→    e.dist = (best / (this.scene3d.pathPoints.length - 1)) * this.scene3d.pathLength;
   406→    this.enemies.push(e);
   407→    this.vfx.burst(pos.clone().add(new THREE.Vector3(0, 1, 0)), data.accent, 14, { speed: 5, life: 0.6, size: 0.6, gravity: 3 });
   408→  }

   410→  /** سپر اثری آریا — مصونیت کامل هسته */
   411→  private requestCoreShield(duration: number) {
   412→    this.shieldUntil = Math.max(this.shieldUntil, this.time + duration);
   413→    this.bus.emit('shield', { active: true, duration });
   414→  }

   416→  private handleKill(e: Enemy) {
   417→    this.coins += e.reward;
   418→    this.coinsEarned += e.reward;
   419→    this.crystals += e.data.crystalReward ?? 0;
   420→    this.killsThisLevel++;
   421→    this.save.stats.kills++;
   422→    if (e.data.boss) {
   423→      this.save.stats.bossKills++;
   424→      this.audio.setBossMode(false);
   425→    }
   426→    this.hero.gainXp(e.data.boss ? 200 : 12);
   427→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   428→  }
   429→
   430→  private handleLeak(e: Enemy) {
   431→    // حذف از صحنه
   432→    e.alive = false;
   433→    e.remove();
   434→    const i = this.enemies.indexOf(e);
   435→    if (i >= 0) this.enemies.splice(i, 1);
   436→
   437→    if (this.time < this.shieldUntil) {
   438→      // سپر آریانا — جلوگیری کامل
   439→      this.vfx.ring(this.scene3d.corePos, 0xfbbf24, 6, 0.5);
   440→      return;
   441→    }
   442→    this.leaks++;
   443→    this.coreHp = Math.max(0, this.coreHp - e.data.coreDamage);
   444→    this.cameraCtrl.shake(e.data.boss ? 1 : 0.45);
   445→    this.vfx.explosion(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 2, 0)), 0xef4444, e.data.boss);
   446→    this.audio.play('explosion');
   447→    this.bus.emit('core', { hp: this.coreHp, max: this.coreMaxHp });
   448→    if (this.coreHp <= 0 && !this.gameOver) {
   449→      this.doDefeat();
   450→    }
   451→  }

   455→  /** از کار انداختن موقت برج توسط توانایی فاز ۲ Boss */
   456→  disableTower(towerId: number, duration: number, color = 0xf97316) {
   457→    const t = this.towers.find((x) => x.id === towerId);
   458→    if (!t || !t.alive) return;
   459→    t.disable(duration, color);

   462→  setBuildMode(towerId: string | null) {
   463→    // قفل برج — باز شدن تدریجی با ستاره
   464→    if (towerId && !isTowerUnlocked(this.save, towerId)) {
   465→      this.audio.play('error');

   578→  tryBuild(towerId: string, point: THREE.Vector3): boolean {
   579→    const data = TOWERS[towerId];
   580→    if (!data) return false;
   581→    // قفل برج — حتی با تپ مستقیم روی زمین
   582→    if (!isTowerUnlocked(this.save, towerId)) {
   583→      this.audio.play('error');
   584→      this.bus.emit('toast', { key: 'towerLocked', type: 'error' });
   585→      return false;
   586→    }
   587→    const x = Math.round(point.x / 2) * 2;
   588→    const z = Math.round(point.z / 2) * 2;
   589→    if (!this.scene3d.canBuild(x, z, this.towers.map((t) => t.position))) {
   590→      this.audio.play('error');
   591→      this.bus.emit('toast', { key: 'invalidSpot', type: 'error' });
   592→      return false;
   593→    }
   594→    if (this.coins < data.cost) {
   595→      this.audio.play('error');
   596→      this.bus.emit('toast', { key: 'notEnoughCoins', type: 'error' });
   597→      return false;
   598→    }
   599→    this.coins -= data.cost;
   600→    const tower = new Tower(this.towerNextId++, towerId, new THREE.Vector3(x, 0, z), this.scene3d.scene, this.ctx);
   601→    // آهنگری آذر — تقویت دائمی این برج از ذخیره
   602→    tower.metaPct = towerMetaPct(towerMetaLevel(this.save, towerId));
   603→    this.towers.push(tower);
   604→    this.towersBuilt++;
   605→    this.save.stats.towersBuilt++;
   606→    this.audio.play('place');
   607→    this.vfx.ring(tower.position, data.color, 4, 0.5);
   608→    this.bus.emit('resources', { coins: this.coins, crystals: this.crystals });
   609→    // خروج از حالت ساخت — هماهنگ با رویداد buildMode
   610→    this.buildMode = null;
   611→    this.clearGhost();
   612→    this.setBuildGridVisible(false);
   613→    this.bus.emit('buildMode', { towerId: null });
   614→    if (!this.save.tutorialDone) this.bus.emit('tutorial', 'wave');
   615→    return true;
   616→  }

   832→  private chainLightning(from: THREE.Vector3, first: DamageableEnemy, chains: number, damage: number, color: number) {
   833→    const hitList: DamageableEnemy[] = [];
   834→    let current = first;
   835→    let currentDamage = damage;
   836→    let currentFrom = from;
   837→    for (let i = 0; i <= chains; i++) {
   838→      if (!current || !current.alive) break;
   839→      this.vfx.bolt(currentFrom, current.getPos().clone().add(new THREE.Vector3(0, 1.2, 0)), color);
   840→      hitList.push(current);
   841→      current.takeDamage(currentDamage, 'energy');
   842→      currentDamage *= 0.72;
   843→      currentFrom = current.getPos().clone().add(new THREE.Vector3(0, 1, 0));
   844→      // هدف بعدی
   845→      let next: DamageableEnemy | null = null;
   846→      let bestD = 8;
   847→      for (const e of this.enemies) {
   848→        if (!e.alive || hitList.includes(e)) continue;
   849→        const d = e.getPos().distanceTo(currentFrom);
   850→        if (d < bestD) {
   851→          bestD = d;
   852→          next = e;
   853→        }
   854→      }
   855→      current = next as DamageableEnemy;
   856→      if (!next) break;
   857→    }
   858→  }

   875→  private handleTap(hit: TapHit) {
   876→    if (this.gameOver) return;
   877→
   878→    // انتخاب شیء؟
   879→    if (hit.objects.length > 0) {
   880→      const root = hit.objects[0].object;
   881→      // برج؟
   882→      const tower = this.towers.find((t) => {
   883→        let o: THREE.Object3D | null = root;
   884→        while (o) {
   885→          if (o === t.mesh) return true;
   886→          o = o.parent;
   887→        }
   888→        return false;
   889→      });
   890→      if (tower) {
   891→        this.selectedTower = tower;
   892→        this.emitTowerInfo();
   893→        this.buildMode = null;
   894→        this.heroSelectMode = false;
   895→        this.clearGhost();
   896→        this.setBuildGridVisible(false);
   897→        this.audio.play('tap');
   898→        return;
   899→      }
   900→      // قهرمان؟ — تپ روی هیرو = ورود به حالت جابجایی
   901→      if (root === this.hero.mesh || (root.parent && root.parent === this.hero.mesh)) {
   902→        if (this.hero.alive) {
   903→          this.buildMode = null;
   904→          this.clearGhost();
   905→          this.setBuildGridVisible(false);
   906→          this.selectHeroMode(true);
   907→          this.bus.emit('heroSelected', {});
   908→          this.bus.emit('toast', { key: 'heroMoveHint', type: 'info' });
   909→        } else {
   910→          this.bus.emit('heroSelected', {});
   911→        }
   912→        this.audio.play('tap');
   913→        return;
   914→      }
   915→    }
   916→
   917→    // جادو در انتظار هدف
   918→    if (this.magicPending) {
   919→      const id = this.magicPending;
   920→      this.magicPending = null;
   921→      this.heroSelectMode = false;
   922→      this.bus.emit('magicPending', { id: null });
   923→      this.castMagic(id, hit.ground);
   924→      return;
   925→    }
   926→
   927→    // حالت ساخت — اول پیش‌نمایش در نقطه تپ، بعد ساخت
   928→    if (this.buildMode) {
   929→      this.updateGhost(hit.ground);
   930→      this.tryBuild(this.buildMode, hit.ground);
   931→      return;
   932→    }
   933→
   934→    // حرکت/استقرار قهرمان
   935→    if (this.heroSelectMode) {
   936→      if (!this.hero.deployed) {
   937→        this.deployHero(hit.ground);
   938→      } else {
   939→        this.hero.moveTo(hit.ground);
   940→        this.vfx.ring(hit.ground, 0xffb020, 1.5, 0.4);
   941→      }
   942→      this.selectHeroMode(false);
   943→      return;

   966→  private doVictory() {
   967→    this.gameOver = true;
   968→    const healthPct = (this.coreHp / this.coreMaxHp) * 100;
   969→    const star1 = true;
   970→    const star2 = healthPct >= this.level.star2HealthPct;
   971→    const star3 = this.leaks === 0;
   972→    const stars = (star1 ? 1 : 0) + (star2 ? 1 : 0) + (star3 ? 1 : 0);
   973→
   974→    // پاداش‌ها — دانش باستانی با عمق کمپین اوج می‌گیرد + پاداش نخستین فتح
   975→    const prev = this.save.levels[this.level.id];
   976→    const firstClear = !prev?.completed;
   977→    const rewardCrystals = 10 + stars * 5;
   978→    const rewardKnowledge = 2 + stars + (this.level.region - 1) * 2 + (firstClear ? this.level.region * 5 : 0);
   979→    this.crystals += rewardCrystals;
   980→    this.save.crystals = this.crystals;
   981→    this.save.knowledge += rewardKnowledge;
   982→    this.save.stats.victories++;
   983→    if (this.leaks === 0 && !this.save.achievements.includes('untouchable')) this.save.achievements.push('untouchable');
   984→    if (this.coreHp === this.coreMaxHp && !this.save.achievements.includes('perfect_defense')) this.save.achievements.push('perfect_defense');
   985→    if (healthPct < 10 && !this.save.achievements.includes('last_stand')) this.save.achievements.push('last_stand');
   986→
   987→    const bestStars = Math.max(prev?.stars ?? 0, stars);
   988→    this.save.levels[this.level.id] = {
   989→      stars: bestStars,
   990→      bestHealthPct: Math.max(prev?.bestHealthPct ?? 0, Math.round(healthPct)),
   991→      completed: true,
   992→    };
   993→    // بازمحاسبه مجموع ستاره‌ها
   994→    this.save.stats.totalStars = Object.values(this.save.levels).reduce((sum, l) => sum + l.stars, 0);
   995→    this.save.tutorialDone = true;
   996→
   997→    this.audio.play('victory');
   998→    this.vfx.burst(this.scene3d.corePos.clone().add(new THREE.Vector3(0, 3, 0)), 0xfbbf24, 80, { speed: 10, life: 1.4, size: 0.8, gravity: 4, up: 1 });
   999→    this.bus.emit('victory', {
  1000→      stars,
  1001→      healthPct: Math.round(healthPct),
  1002→      rewards: { crystals: rewardCrystals, knowledge: rewardKnowledge, coins: this.coinsEarned },
  1003→    });
  1004→  }
